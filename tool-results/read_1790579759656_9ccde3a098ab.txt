    86→    86→function LigneInvocation({ ligne, accent }: { ligne: string; accent: string }) {
    87→    87→  const m = ligne.match(/^(.*?)(\s*[×x]\s*[٠-٩\dHh '’]+)$/u)
    88→    88→  if (m && m[2].trim()) {
    89→    89→    return (
    90→    90→      <span className="block">
    91→    91→        {m[1]}
    92→    92→        <span
    93→    93→          className="mx-1 inline-block rounded-md px-2 py-0.5 align-middle text-[0.72em] font-bold tracking-wide"
    94→    94→          style={{ color: accent, backgroundColor: `${accent}18` }}
    95→    95→        >
    96→    96→          {m[2].trim()}
    97→    97→        </span>
    98→    98→      </span>
    99→    99→    )
   100→   100→  }
   101→   101→  return <span className="block">{ligne || '\u00A0'}</span>
   102→   102→}

   113→   113→function LecteurAdhkar({ doc, langue, chapitreInitial, onFermer }: LecteurProps) {
   114→   114→  const l = useLecture()
   115→   115→  const imprimer = usePrint((s) => s.imprimer)
   116→   116→
   117→   117→  /* Le diwân de samâ‘ (398 chants) n'entre pas dans le fil principal :
   118→   118→     il se charge paresseusement à l'ouverture du document. */
   119→   119→  const estSama = doc.id === 'sama' && langue === 'ph'
   120→   120→  const [samaTexte, setSamaTexte] = useState<ChapitreLecture[] | null>(null)
   121→   121→  useEffect(() => {
   122→   122→    if (doc.id !== 'sama' || langue !== 'ph') return
   123→   123→    let annule = false
   124→   124→    import('@/data/sama-chants').then((m) => {
   125→   125→      if (!annule) setSamaTexte(m.CHAPITRES_SAMA)
   126→   126→    })
   127→   127→    return () => {
   128→   128→      annule = true
   129→   129→    }
   130→   130→  }, [doc.id, langue])
   131→   131→
   132→   132→  const chapitres = useMemo(
   133→   133→    () => (doc.id === 'sama' && langue === 'ph' ? samaTexte ?? [] : chapitresDuDoc(doc, langue)),
   134→   134→    [doc, langue, samaTexte]
   135→   135→  )
   136→   136→  const pdf = pdfDuDoc(doc, langue)
   137→   137→  const conteneurRef = useRef<HTMLDivElement | null>(null)
   138→   138→  const cartesRef = useRef<(HTMLElement | null)[]>([])
   139→   139→
   140→   140→  const [indexActif, setIndexActif] = useState(() => {
   141→   141→    const i = chapitreInitial ? chapitres.findIndex((c) => c.id === chapitreInitial) : -1
   142→   142→    return i >= 0 ? i : 0
   143→   143→  })
   144→   144→  const [reglagesOuverts, setReglagesOuverts] = useState(false)
   145→   145→  const [sommaireOuvert, setSommaireOuvert] = useState(false)
   146→   146→  const [progression, setProgression] = useState(0)
   147→   147→  const [scrolle, setScrolle] = useState(false)
   148→   148→
   149→   149→  const arabe = langue === 'ar'
   150→   150→  /* Vue du lecteur : le TEXTE RÉÉCRIT d'abord — encre sur porcelaine,
   151→   151→     polices et couleurs réglables, évolutif. La page numérisée reste
   152→   152→     disponible pour lire le livre imprimé à l'identique. La recherche ⌘K
   153→   153→     ouvre directement le texte au chapitre demandé. */
   154→   154→  const aLesDeux = Boolean(pdf) && chapitres.length > 0
   155→   155→  const [vue, setVue] = useState<'doc' | 'texte'>(
   156→   156→    chapitreInitial || !pdf || chapitres.length > 0 || estSama ? 'texte' : 'doc'
   157→   157→  )
   158→   158→  const vueDoc = vue === 'doc' && Boolean(pdf)
   159→   159→  const theme = THEMES.find((t) => t.id === l.theme) ?? THEMES[0]
   160→   160→  const accent = ACCENTS.find((a) => a.id === l.accent) ?? ACCENTS[0]
   161→   161→  const police = arabe
   162→   162→    ? POLICES_AR.find((p) => p.id === l.policeArabe) ?? POLICES_AR[0]
   163→   163→    : POLICES_PH.find((p) => p.id === l.police) ?? POLICES_PH[0]
   164→   164→  const taille = arabe ? l.tailleArabe : l.taille
   165→   165→
   166→   166→  /* ─── L'écran ne s'éteint pas pendant qu'on lit ───
   167→   167→     Un frère qui récite ne touche pas son téléphone pendant de
   168→   168→     longues minutes. Le verrou d'écran est demandé à l'ouverture,
   169→   169→     rendu si la page passe à l'arrière-plan, et relâché après un
   170→   170→     quart d'heure sans le moindre geste — un téléphone oublié
   171→   171→     écran allumé chauffe. Tout échec est muet. */
   172→   172→  useEffect(() => {
   173→   173→    if (!('wakeLock' in navigator)) return
   174→   174→    let verrou: WakeLockSentinel | null = null
   175→   175→    let minuteur: number | undefined
   176→   176→    const REPOS = 15 * 60 * 1000
   177→   177→
   178→   178→    const demander = async () => {
   179→   179→      try {
   180→   180→        verrou = await navigator.wakeLock.request('screen')
   181→   181→      } catch {
   182→   182→        /* appareil qui ne sait pas faire : on lit comme avant */
   183→   183→      }
   184→   184→    }
   185→   185→    const relacher = () => {
   186→   186→      verrou?.release().catch(() => undefined)
   187→   187→      verrou = null
   188→   188→    }
   189→   189→    const toucher = () => {
   190→   190→      window.clearTimeout(minuteur)
   191→   191→      minuteur = window.setTimeout(relacher, REPOS)
   192→   192→    }
   193→   193→    const surVisibilite = () => {
   194→   194→      if (document.visibilityState === 'visible') {
   195→   195→        demander()
   196→   196→        toucher()
   197→   197→      } else {
   198→   198→        relacher()
   199→   199→      }
   200→   200→    }
   201→   201→    demander()
   202→   202→    toucher()
   203→   203→    document.addEventListener('visibilitychange', surVisibilite)
   204→   204→    window.addEventListener('pointerdown', toucher, { passive: true })
   205→   205→    return () => {
   206→   206→      relacher()
   207→   207→      window.clearTimeout(minuteur)
   208→   208→      document.removeEventListener('visibilitychange', surVisibilite)
   209→   209→      window.removeEventListener('pointerdown', toucher)
   210→   210→    }
   211→   211→  }, [])
   212→   212→
   213→   213→  /* ─── Le corps de la page ne défile pas derrière le lecteur ─── */
   214→   214→  useEffect(() => {
   215→   215→    const deplacement = window.scrollY
   216→   216→    document.body.style.overflow = 'hidden'
   217→   217→    return () => {
   218→   218→      document.body.style.overflow = ''
   219→   219→      window.scrollTo(0, deplacement)
   220→   220→    }
   221→   221→  }, [])
   222→   222→
   223→   223→  /* ─── Échap ferme le lecteur ─── */
   224→   224→  useEffect(() => {
   225→   225→    const surTouche = (e: KeyboardEvent) => {
   226→   226→      if (e.key === 'Escape') onFermer()
   227→   227→    }
   228→   228→    window.addEventListener('keydown', surTouche)
   229→   229→    return () => window.removeEventListener('keydown', surTouche)
   230→   230→  }, [onFermer])
   231→   231→
   232→   232→  /* ─── Progression de la lecture + affichage du bouton revenir ─── */
   233→   233→  useEffect(() => {
   234→   234→    const el = conteneurRef.current
   235→   235→    if (!el) return
   236→   236→    const surScroll = () => {
   237→   237→      const reste = el.scrollHeight - el.clientHeight
   238→   238→      setProgression(reste > 0 ? Math.min(1, el.scrollTop / reste) : 0)
   239→   239→      setScrolle(el.scrollTop > 260)
   240→   240→    }
   241→   241→    el.addEventListener('scroll', surScroll, { passive: true })
   242→   242→    surScroll()
   243→   243→    return () => el.removeEventListener('scroll', surScroll)
   244→   244→  }, [vueDoc])
   245→   245→
   246→   246→  /* ─── Positionner au chapitre demandé à l'ouverture ───
   247→   247→     (le diwân arrive un instant après : on replace alors) */
   248→   248→  useEffect(() => {
   249→   249→    const el = conteneurRef.current
   250→   250→    if (!el) return
   251→   251→    if (chapitreInitial) {
   252→   252→      const i = chapitres.findIndex((c) => c.id === chapitreInitial)
   253→   253→      const cible = i >= 0 ? cartesRef.current[i] : null
   254→   254→      if (cible) {
   255→   255→        el.scrollTo({ top: cible.offsetTop - el.offsetTop - 8 })
   256→   256→        return
   257→   257→      }
   258→   258→    }
   259→   259→    el.scrollTo({ top: 0 })
   260→   260→  }, [chapitreInitial, langue, doc.id, chapitres.length])
   261→   261→
   262→   262→  const allerAuChapitre = useCallback(
   263→   263→    (i: number) => {
   264→   264→      const cible = cartesRef.current[i]
   265→   265→      if (cible) {
   266→   266→        conteneurRef.current?.scrollTo({
   267→   267→          top: cible.offsetTop - (conteneurRef.current?.offsetTop ?? 0) - 8,
   268→   268→          behavior: 'smooth',
   269→   269→        })
   270→   270→      }
   271→   271→    },
   272→   272→    []
   273→   273→  )
   274→   274→
   275→   275→  /* ─── Le chapitre visible suit le défilement ─── */
   276→   276→  useEffect(() => {
   277→   277→    if (chapitres.length < 2) return
   278→   278→    const el = conteneurRef.current
   279→   279→    if (!el) return
   280→   280→    const observateur = new IntersectionObserver(
   281→   281→      (entrees) => {
   282→   282→        for (const e of entrees) {
   283→   283→          if (e.isIntersecting) {
   284→   284→            const i = Number((e.target as HTMLElement).dataset.i)
   285→   285→            if (!Number.isNaN(i)) setIndexActif(i)
   286→   286→          }
   287→   287→        }
   288→   288→      },
   289→   289→      { root: el, rootMargin: '-15% 0px -60% 0px', threshold: 0 }
   290→   290→    )
   291→   291→    cartesRef.current.forEach((c) => c && observateur.observe(c))
   292→   292→    return () => observateur.disconnect()
   293→   293→  }, [chapitres.length, langue, doc.id])
   294→   294→
   295→   295→  const titre = arabe ? doc.titreAr : doc.titreFr
   296→   296→
   297→   297→  return (
   298→   298→    <motion.div
   299→   299→      initial={{ opacity: 0 }}
   300→   300→      animate={{ opacity: 1 }}
   301→   301→      exit={{ opacity: 0 }}
   302→   302→      transition={{ duration: 0.22 }}
   303→   303→      className="fixed inset-0 z-[90] flex flex-col bg-background grain-papier"
   304→   304→      role="dialog"
   305→   305→      aria-modal="true"
   306→   306→      aria-label={`Lecteur — ${titre}`}
   307→   307→    >
   308→   308→      {/* ─── La barre du haut : titre, sommaire, réglages, fermeture ─── */}
   309→   309→      <header className="relative z-10 flex-none border-b border-border bg-card/95 backdrop-blur supports-[backdrop-filter]:bg-card/90">
   310→   310→        <div
   311→   311→          className="flex items-center gap-1 px-2.5"
   312→   312→          style={{ paddingTop: 'calc(10px + env(safe-area-inset-top))', paddingBottom: 10 }}
   313→   313→        >
   314→   314→          {/* Réglages de lecture — le texte seulement */}
   315→   315→          {!vueDoc && (
   316→   316→            <button
   317→   317→              type="button"
   318→   318→              onClick={() => {
   319→   319→                setReglagesOuverts((o) => !o)
   320→   320→                setSommaireOuvert(false)
   321→   321→              }}
   322→   322→              aria-expanded={reglagesOuverts}
   323→   323→              aria-label="Réglages de lecture"
   324→   324→              className={cn(
   325→   325→                'grid h-9 w-9 place-items-center rounded-full transition-colors duration-200',
   326→   326→                reglagesOuverts
   327→   327→                  ? 'bg-muted text-foreground'
   328→   328→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   329→   329→              )}
   330→   330→            >
   331→   331→              <Type className="h-4.5 w-4.5" />
   332→   332→            </button>
   333→   333→          )}
   334→   334→
   335→   335→          {/* Sommaire des chapitres */}
   336→   336→          {!vueDoc && chapitres.length > 1 && (
   337→   337→            <button
   338→   338→              type="button"
   339→   339→              onClick={() => {
   340→   340→                setSommaireOuvert((o) => !o)
   341→   341→                setReglagesOuverts(false)
   342→   342→              }}
   343→   343→              aria-expanded={sommaireOuvert}
   344→   344→              aria-label="Sommaire du document"
   345→   345→              className={cn(
   346→   346→                'grid h-9 w-9 place-items-center rounded-full transition-colors duration-200',
   347→   347→                sommaireOuvert
   348→   348→                  ? 'bg-muted text-foreground'
   349→   349→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   350→   350→              )}
   351→   351→            >
   352→   352→              <List className="h-4.5 w-4.5" />
   353→   353→            </button>
   354→   354→          )}
   355→   355→
   356→   356→          {/* Le titre, centré — le nom de la pièce */}
   357→   357→          <h2
   358→   358→            className={cn(
   359→   359→              'min-w-0 flex-1 truncate text-center text-[16px] font-medium text-encre',
   360→   360→              arabe && 'font-arabic text-[19px]'
   361→   361→            )}
   362→   362→            dir={arabe ? 'rtl' : undefined}
   363→   363→            title={titre}
   364→   364→          >
   365→   365→            {titre}
   366→   366→          </h2>
   367→   367→
   368→   368→          {/* Fermeture */}
   369→   369→          <button
   370→   370→            type="button"
   371→   371→            onClick={onFermer}
   372→   372→            aria-label="Fermer le lecteur"
   373→   373→            className="grid h-9 w-9 place-items-center rounded-full border border-border text-brouillard transition-colors duration-200 hover:bg-muted/60 hover:text-foreground active:bg-muted"
   374→   374→          >
   375→   375→            <X className="h-4.5 w-4.5" />
   376→   376→          </button>
   377→   377→        </div>
   378→   378→
   379→   379→        {/* Filet de progression de la lecture */}
   380→   380→        <div className="absolute inset-x-0 bottom-0 h-[2px] bg-border" aria-hidden>
   381→   381→          <div
   382→   382→            className="h-full bg-primary transition-[width] duration-150"
   383→   383→            style={{ width: `${Math.round(progression * 100)}%` }}
   384→   384→          />
   385→   385→        </div>
   386→   386→
   387→   387→        {/* Bascule Texte / Numérisé — le texte réécrit d'abord (encre
   388→   388→            sur porcelaine), ou le livre imprimé à l'identique */}
   389→   389→        {aLesDeux && (
   390→   390→          <div className="flex justify-center gap-1.5 border-t border-border py-1.5">
   391→   391→            <button
   392→   392→              type="button"
   393→   393→              onClick={() => {
   394→   394→                setVue('texte')
   395→   395→                setReglagesOuverts(false)
   396→   396→                setSommaireOuvert(false)
   397→   397→              }}
   398→   398→              aria-pressed={!vueDoc}
   399→   399→              className={cn(
   400→   400→                'flex items-center gap-1.5 rounded-full px-3.5 py-1 text-[11px] font-medium uppercase tracking-[0.14em] transition-colors duration-200 active:scale-[0.98]',
   401→   401→                !vueDoc
   402→   402→                  ? 'bg-primary text-primary-foreground'
   403→   403→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   404→   404→              )}
   405→   405→            >
   406→   406→              <Type className="h-3.5 w-3.5" /> Texte
   407→   407→            </button>
   408→   408→            <button
   409→   409→              type="button"
   410→   410→              onClick={() => {
