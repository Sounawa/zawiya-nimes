    85→/* ─── Une ligne du texte, répétitions « x 100 » / « × ١٠٠ » en badge ─── */
    86→function LigneInvocation({ ligne, accent }: { ligne: string; accent: string }) {
    87→  const m = ligne.match(/^(.*?)(\s*[×x]\s*[٠-٩\dHh '’]+)$/u)
    88→  if (m && m[2].trim()) {
    89→    return (
    90→      <span className="block">
    91→        {m[1]}
    92→        <span
    93→          className="mx-1 inline-block rounded-md px-2 py-0.5 align-middle text-[0.72em] font-bold tracking-wide"
    94→          style={{ color: accent, backgroundColor: `${accent}18` }}
    95→        >
    96→          {m[2].trim()}
    97→        </span>
    98→      </span>
    99→    )
   100→  }
   101→  return <span className="block">{ligne || '\u00A0'}</span>
   102→}

   113→function LecteurAdhkar({ doc, langue, chapitreInitial, onFermer }: LecteurProps) {
   114→  const l = useLecture()
   115→  const imprimer = usePrint((s) => s.imprimer)
   116→
   117→  /* Le diwân de samâ‘ (398 chants) n'entre pas dans le fil principal :
   118→     il se charge paresseusement à l'ouverture du document. */
   119→  const estSama = doc.id === 'sama' && langue === 'ph'
   120→  const [samaTexte, setSamaTexte] = useState<ChapitreLecture[] | null>(null)
   121→  useEffect(() => {
   122→    if (doc.id !== 'sama' || langue !== 'ph') return
   123→    let annule = false
   124→    import('@/data/sama-chants').then((m) => {
   125→      if (!annule) setSamaTexte(m.CHAPITRES_SAMA)
   126→    })
   127→    return () => {
   128→      annule = true
   129→    }
   130→  }, [doc.id, langue])
   131→
   132→  const chapitres = useMemo(
   133→    () => (doc.id === 'sama' && langue === 'ph' ? samaTexte ?? [] : chapitresDuDoc(doc, langue)),
   134→    [doc, langue, samaTexte]
   135→  )
   136→  const pdf = pdfDuDoc(doc, langue)
   137→  const conteneurRef = useRef<HTMLDivElement | null>(null)
   138→  const cartesRef = useRef<(HTMLElement | null)[]>([])
   139→
   140→  const [indexActif, setIndexActif] = useState(() => {
   141→    const i = chapitreInitial ? chapitres.findIndex((c) => c.id === chapitreInitial) : -1
   142→    return i >= 0 ? i : 0
   143→  })
   144→  const [reglagesOuverts, setReglagesOuverts] = useState(false)
   145→  const [sommaireOuvert, setSommaireOuvert] = useState(false)
   146→  const [progression, setProgression] = useState(0)
   147→  const [scrolle, setScrolle] = useState(false)
   148→
   149→  const arabe = langue === 'ar'
   150→  /* Vue du lecteur : le TEXTE RÉÉCRIT d'abord — encre sur porcelaine,
   151→     polices et couleurs réglables, évolutif. La page numérisée reste
   152→     disponible pour lire le livre imprimé à l'identique. La recherche ⌘K
   153→     ouvre directement le texte au chapitre demandé. */
   154→  const aLesDeux = Boolean(pdf) && chapitres.length > 0
   155→  const [vue, setVue] = useState<'doc' | 'texte'>(
   156→    chapitreInitial || !pdf || chapitres.length > 0 || estSama ? 'texte' : 'doc'
   157→  )
   158→  const vueDoc = vue === 'doc' && Boolean(pdf)
   159→  const theme = THEMES.find((t) => t.id === l.theme) ?? THEMES[0]
   160→  const accent = ACCENTS.find((a) => a.id === l.accent) ?? ACCENTS[0]
   161→  const police = arabe
   162→    ? POLICES_AR.find((p) => p.id === l.policeArabe) ?? POLICES_AR[0]
   163→    : POLICES_PH.find((p) => p.id === l.police) ?? POLICES_PH[0]
   164→  const taille = arabe ? l.tailleArabe : l.taille
   165→
   166→  /* ─── L'écran ne s'éteint pas pendant qu'on lit ───
   167→     Un frère qui récite ne touche pas son téléphone pendant de
   168→     longues minutes. Le verrou d'écran est demandé à l'ouverture,
   169→     rendu si la page passe à l'arrière-plan, et relâché après un
   170→     quart d'heure sans le moindre geste — un téléphone oublié
   171→     écran allumé chauffe. Tout échec est muet. */
   172→  useEffect(() => {
   173→    if (!('wakeLock' in navigator)) return
   174→    let verrou: WakeLockSentinel | null = null
   175→    let minuteur: number | undefined
   176→    const REPOS = 15 * 60 * 1000
   177→
   178→    const demander = async () => {
   179→      try {
   180→        verrou = await navigator.wakeLock.request('screen')
   181→      } catch {
   182→        /* appareil qui ne sait pas faire : on lit comme avant */
   183→      }
   184→    }
   185→    const relacher = () => {
   186→      verrou?.release().catch(() => undefined)
   187→      verrou = null
   188→    }
   189→    const toucher = () => {
   190→      window.clearTimeout(minuteur)
   191→      minuteur = window.setTimeout(relacher, REPOS)
   192→    }
   193→    const surVisibilite = () => {
   194→      if (document.visibilityState === 'visible') {
   195→        demander()
   196→        toucher()
   197→      } else {
   198→        relacher()
   199→      }
   200→    }
   201→    demander()
   202→    toucher()
   203→    document.addEventListener('visibilitychange', surVisibilite)
   204→    window.addEventListener('pointerdown', toucher, { passive: true })
   205→    return () => {
   206→      relacher()
   207→      window.clearTimeout(minuteur)
   208→      document.removeEventListener('visibilitychange', surVisibilite)
   209→      window.removeEventListener('pointerdown', toucher)
   210→    }
   211→  }, [])
   212→
   213→  /* ─── Le corps de la page ne défile pas derrière le lecteur ─── */
   214→  useEffect(() => {
   215→    const deplacement = window.scrollY
   216→    document.body.style.overflow = 'hidden'
   217→    return () => {
   218→      document.body.style.overflow = ''
   219→      window.scrollTo(0, deplacement)
   220→    }
   221→  }, [])
   222→
   223→  /* ─── Échap ferme le lecteur ─── */
   224→  useEffect(() => {
   225→    const surTouche = (e: KeyboardEvent) => {
   226→      if (e.key === 'Escape') onFermer()
   227→    }
   228→    window.addEventListener('keydown', surTouche)
   229→    return () => window.removeEventListener('keydown', surTouche)
   230→  }, [onFermer])
   231→
   232→  /* ─── Progression de la lecture + affichage du bouton revenir ─── */
   233→  useEffect(() => {
   234→    const el = conteneurRef.current
   235→    if (!el) return
   236→    const surScroll = () => {
   237→      const reste = el.scrollHeight - el.clientHeight
   238→      setProgression(reste > 0 ? Math.min(1, el.scrollTop / reste) : 0)
   239→      setScrolle(el.scrollTop > 260)
   240→    }
   241→    el.addEventListener('scroll', surScroll, { passive: true })
   242→    surScroll()
   243→    return () => el.removeEventListener('scroll', surScroll)
   244→  }, [vueDoc])
   245→
   246→  /* ─── Positionner au chapitre demandé à l'ouverture ───
   247→     (le diwân arrive un instant après : on replace alors) */
   248→  useEffect(() => {
   249→    const el = conteneurRef.current
   250→    if (!el) return
   251→    if (chapitreInitial) {
   252→      const i = chapitres.findIndex((c) => c.id === chapitreInitial)
   253→      const cible = i >= 0 ? cartesRef.current[i] : null
   254→      if (cible) {
   255→        el.scrollTo({ top: cible.offsetTop - el.offsetTop - 8 })
   256→        return
   257→      }
   258→    }
   259→    el.scrollTo({ top: 0 })
   260→  }, [chapitreInitial, langue, doc.id, chapitres.length])
   261→
   262→  const allerAuChapitre = useCallback(
   263→    (i: number) => {
   264→      const cible = cartesRef.current[i]
   265→      if (cible) {
   266→        conteneurRef.current?.scrollTo({
   267→          top: cible.offsetTop - (conteneurRef.current?.offsetTop ?? 0) - 8,
   268→          behavior: 'smooth',
   269→        })
   270→      }
   271→    },
   272→    []
   273→  )
   274→
   275→  /* ─── Le chapitre visible suit le défilement ─── */
   276→  useEffect(() => {
   277→    if (chapitres.length < 2) return
   278→    const el = conteneurRef.current
   279→    if (!el) return
   280→    const observateur = new IntersectionObserver(
   281→      (entrees) => {
   282→        for (const e of entrees) {
   283→          if (e.isIntersecting) {
   284→            const i = Number((e.target as HTMLElement).dataset.i)
   285→            if (!Number.isNaN(i)) setIndexActif(i)
   286→          }
   287→        }
   288→      },
   289→      { root: el, rootMargin: '-15% 0px -60% 0px', threshold: 0 }
   290→    )
   291→    cartesRef.current.forEach((c) => c && observateur.observe(c))
   292→    return () => observateur.disconnect()
   293→  }, [chapitres.length, langue, doc.id])
   294→
   295→  const titre = arabe ? doc.titreAr : doc.titreFr
   296→
   297→  return (
   298→    <motion.div
   299→      initial={{ opacity: 0 }}
   300→      animate={{ opacity: 1 }}
   301→      exit={{ opacity: 0 }}
   302→      transition={{ duration: 0.22 }}
   303→      /* mb-0 : parade contre les space-y-* d'un ancêtre (Tailwind v4
   304→          pose margin-block-end sur :not(:last-child)) — un lecteur plein
   305→          écran ne doit jamais recevoir de marge, où qu'il vive */
   306→      className="fixed inset-0 z-[90] mb-0 flex flex-col bg-background grain-papier"
   307→      role="dialog"
   308→      aria-modal="true"
   309→      aria-label={`Lecteur — ${titre}`}
   310→    >
   311→      {/* ─── La barre du haut : titre, sommaire, réglages, fermeture ─── */}
   312→      <header className="relative z-10 flex-none border-b border-border bg-card/95 backdrop-blur supports-[backdrop-filter]:bg-card/90">
   313→        <div
   314→          className="flex items-center gap-1 px-2.5"
   315→          style={{ paddingTop: 'calc(10px + env(safe-area-inset-top))', paddingBottom: 10 }}
   316→        >
   317→          {/* Réglages de lecture — le texte seulement */}
   318→          {!vueDoc && (
   319→            <button
   320→              type="button"
   321→              onClick={() => {
   322→                setReglagesOuverts((o) => !o)
   323→                setSommaireOuvert(false)
   324→              }}
   325→              aria-expanded={reglagesOuverts}
   326→              aria-label="Réglages de lecture"
   327→              className={cn(
   328→                'grid h-9 w-9 place-items-center rounded-full transition-colors duration-200',
   329→                reglagesOuverts
   330→                  ? 'bg-muted text-foreground'
   331→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   332→              )}
   333→            >
   334→              <Type className="h-4.5 w-4.5" />
   335→            </button>
   336→          )}
   337→
   338→          {/* Sommaire des chapitres */}
   339→          {!vueDoc && chapitres.length > 1 && (
   340→            <button
   341→              type="button"
   342→              onClick={() => {
   343→                setSommaireOuvert((o) => !o)
   344→                setReglagesOuverts(false)
   345→              }}
   346→              aria-expanded={sommaireOuvert}
   347→              aria-label="Sommaire du document"
   348→              className={cn(
   349→                'grid h-9 w-9 place-items-center rounded-full transition-colors duration-200',
   350→                sommaireOuvert
   351→                  ? 'bg-muted text-foreground'
   352→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   353→              )}
   354→            >
   355→              <List className="h-4.5 w-4.5" />
   356→            </button>
   357→          )}
   358→
   359→          {/* Le titre, centré — le nom de la pièce */}
   360→          <h2
   361→            className={cn(
   362→              'min-w-0 flex-1 truncate text-center text-[16px] font-medium text-encre',
   363→              arabe && 'font-arabic text-[19px]'
   364→            )}
   365→            dir={arabe ? 'rtl' : undefined}
   366→            title={titre}
   367→          >
   368→            {titre}
   369→          </h2>
   370→
   371→          {/* Fermeture */}
   372→          <button
   373→            type="button"
   374→            onClick={onFermer}
   375→            aria-label="Fermer le lecteur"
   376→            className="grid h-9 w-9 place-items-center rounded-full border border-border text-brouillard transition-colors duration-200 hover:bg-muted/60 hover:text-foreground active:bg-muted"
   377→          >
   378→            <X className="h-4.5 w-4.5" />
   379→          </button>
   380→        </div>
   381→
   382→        {/* Filet de progression de la lecture */}
   383→        <div className="absolute inset-x-0 bottom-0 h-[2px] bg-border" aria-hidden>
   384→          <div
   385→            className="h-full bg-primary transition-[width] duration-150"
   386→            style={{ width: `${Math.round(progression * 100)}%` }}
   387→          />
   388→        </div>
   389→
   390→        {/* Bascule Texte / Numérisé — le texte réécrit d'abord (encre
   391→            sur porcelaine), ou le livre imprimé à l'identique */}
   392→        {aLesDeux && (
   393→          <div className="flex justify-center gap-1.5 border-t border-border py-1.5">
   394→            <button
   395→              type="button"
   396→              onClick={() => {
   397→                setVue('texte')
   398→                setReglagesOuverts(false)
   399→                setSommaireOuvert(false)
   400→              }}
   401→              aria-pressed={!vueDoc}
   402→              className={cn(
   403→                'flex items-center gap-1.5 rounded-full px-3.5 py-1 text-[11px] font-medium uppercase tracking-[0.14em] transition-colors duration-200 active:scale-[0.98]',
   404→                !vueDoc
   405→                  ? 'bg-primary text-primary-foreground'
   406→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   407→              )}
   408→            >
   409→              <Type className="h-3.5 w-3.5" /> Texte
   410→            </button>
   411→            <button
   412→              type="button"
   413→              onClick={() => {
   414→                setVue('doc')
   415→                setReglagesOuverts(false)
   416→                setSommaireOuvert(false)
   417→              }}
   418→              aria-pressed={vueDoc}
   419→              className={cn(
   420→                'flex items-center gap-1.5 rounded-full px-3.5 py-1 text-[11px] font-medium uppercase tracking-[0.14em] transition-colors duration-200 active:scale-[0.98]',
   421→                vueDoc
   422→                  ? 'bg-primary text-primary-foreground'
   423→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   424→              )}
   425→            >
   426→              <BookOpen className="h-3.5 w-3.5" /> Numérisé
   427→            </button>
   428→          </div>
   429→        )}
   430→
   431→        {/* ─── Panneau des réglages de lecture ─── */}
   432→        <AnimatePresence>
   433→          {reglagesOuverts && (
   434→            <motion.div
   435→              initial={{ height: 0, opacity: 0 }}
   436→              animate={{ height: 'auto', opacity: 1 }}
   437→              exit={{ height: 0, opacity: 0 }}
   438→              transition={{ duration: 0.22 }}
   439→              className="overflow-hidden"
   440→            >
   441→              <div className="space-y-3 border-t border-border bg-card px-4 py-4">
   442→                {/* Police */}
   443→                <div className="flex items-center justify-between gap-3">
   444→                  <span className="text-[10px] font-medium uppercase tracking-[0.3em] text-brouillard">Police</span>
   445→                  <div className="flex gap-1">
   446→                    {(arabe ? POLICES_AR : POLICES_PH).map((p) => (
   447→                      <button
   448→                        key={p.id}
   449→                        type="button"
   450→                        onClick={() =>
   451→                          l.regler(arabe ? { policeArabe: p.id as PoliceArabe } : { police: p.id as PoliceLecture })
   452→                        }
   453→                        aria-pressed={(arabe ? l.policeArabe : l.police) === p.id}
   454→                        className={cn(
   455→                          'rounded-lg border px-3 py-1.5 transition-colors duration-200',
   456→                          arabe ? 'font-arabic text-base' : 'text-sm',
   457→                          (arabe ? l.policeArabe : l.police) === p.id
   458→                            ? 'border-primary/30 bg-primary/10 font-medium text-primary'

   476→                        l.regler(arabe ? { tailleArabe: Math.max(17, l.tailleArabe - 1) } : { taille: Math.max(14, l.taille - 1) })
   477→                      }

   484→                    <button
