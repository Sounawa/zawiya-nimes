    88→/* ─── Une ligne du texte, répétitions « x 100 » / « × ١٠٠ » en badge ─── */
    89→function LigneInvocation({ ligne, accent }: { ligne: string; accent: string }) {
    90→  const m = ligne.match(/^(.*?)(\s*[×x]\s*[٠-٩\dHh '’]+)$/u)
    91→  if (m && m[2].trim()) {
    92→    return (
    93→      <span className="block">
    94→        {m[1]}
    95→        <span
    96→          className="mx-1 inline-block rounded-md px-2 py-0.5 align-middle text-[0.72em] font-bold tracking-wide"
    97→          style={{ color: accent, backgroundColor: `${accent}18` }}
    98→        >
    99→          {m[2].trim()}
   100→        </span>
   101→      </span>
   102→    )
   103→  }
   104→  return <span className="block">{ligne || '\u00A0'}</span>
   105→}

   107→/* ─── Le paragraphage du cahier phonétique ───
   108→   Le cahier imprimé est une colonne étroite : l'extraction garde
   109→   des lignes courtes coupées au milieu des phrases. On rejoint
   110→   les lignes d'un même bloc (séparé par une ligne vide) et on
   111→   coupe après les renvois « → p. » et les répétitions « x 100 » —
   112→   le texte coule alors sur toute la largeur, justifié comme un
   113→   livre, au lieu de rester collé à gauche. */
   114→function paragrapher(texte: string): string[] {
   115→  const paras: string[] = []
   116→  let courant: string[] = []
   117→  const flush = () => {
   118→    if (courant.length) {
   119→      paras.push(courant.join(' ').replace(/\s+/g, ' ').trim())
   120→      courant = []
   121→    }
   122→  }
   123→  for (const brute of texte.split('\n')) {
   124→    const t = brute.trim()
   125→    if (!t) {
   126→      flush()
   127→      continue
   128→    }
   129→    courant.push(t)
   130→    /* Fin logique de paragraphe : renvoi de page, ou répétition badgée */
   131→    if (/→\s*p\.|→\s*ص|[×x]\s*[٠-٩\d][٠-٩\dHh '’]*$/.test(t)) flush()
   132→  }
   133→  flush()
   134→  return paras
   135→}

   156→function LecteurAdhkar({ doc, langue, chapitreInitial, onFermer }: LecteurProps) {
   157→  const l = useLecture()
   158→  const imprimer = usePrint((s) => s.imprimer)
   159→
   160→  /* Le diwân de samâ‘ (398 chants) n'entre pas dans le fil principal :
   161→     il se charge paresseusement à l'ouverture du document. */
   162→  const estSama = doc.id === 'sama' && langue === 'ph'
   163→  const [samaTexte, setSamaTexte] = useState<ChapitreLecture[] | null>(null)
   164→  useEffect(() => {
   165→    if (doc.id !== 'sama' || langue !== 'ph') return
   166→    let annule = false
   167→    import('@/data/sama-chants').then((m) => {
   168→      if (!annule) setSamaTexte(m.CHAPITRES_SAMA)
   169→    })
   170→    return () => {
   171→      annule = true
   172→    }
   173→  }, [doc.id, langue])
   174→
   175→  /* Les entrées du lecteur : pratiques structurées (cahier de
   176→     wadhifa) ou chapitres plats (autres documents). Chaque entrée
   177→     porte sa langue effective — le sommaire ne change jamais quand
   178→     on bascule phonétique ⇄ عربي, la pratique lue reste en place. */
   179→  const entrees = useMemo<EntreeLecture[]>(() => {
   180→    if (doc.pratiques) {
   181→      return doc.pratiques
   182→        .map((p): EntreeLecture | null => {
   183→          const prefere = p[langue]
   184→          const autre = langue === 'ph' ? p.ar : p.ph
   185→          const c = prefere ?? autre
   186→          if (!c) return null
   187→          return {
   188→            ...c,
   189→            langue: prefere ? langue : langue === 'ph' ? 'ar' : 'ph',
   190→            repli: !prefere,
   191→            groupe: p.groupe,
   192→          }
   193→        })
   194→        .filter((e): e is EntreeLecture => e !== null)
   195→    }
   196→    if (doc.id === 'sama' && langue === 'ph') {
   197→      return (samaTexte ?? []).map((c) => ({ ...c, langue: 'ph' as const }))
   198→    }
   199→    return chapitresDuDoc(doc, langue).map((c) => ({ ...c, langue }))
   200→  }, [doc, langue, samaTexte])
   201→  const chapitres = entrees
   202→  const pdf = pdfDuDoc(doc, langue)
   203→  const conteneurRef = useRef<HTMLDivElement | null>(null)
   204→  const cartesRef = useRef<(HTMLElement | null)[]>([])
   205→
   206→  const [indexActif, setIndexActif] = useState(() => {
   207→    const i = chapitreInitial ? entrees.findIndex((c) => c.id === chapitreInitial) : -1
   208→    return i >= 0 ? i : 0
   209→  })
   210→  const [reglagesOuverts, setReglagesOuverts] = useState(false)
   211→  const [sommaireOuvert, setSommaireOuvert] = useState(false)
   212→  /* La carte sommaire du cahier — dépliée à l'ouverture, refermée
   213→     si la recherche ⌘K demande directement une pratique. */
   214→  const [sommaireCarte, setSommaireCarte] = useState(!chapitreInitial)
   215→  const [progression, setProgression] = useState(0)
   216→  const [scrolle, setScrolle] = useState(false)
   217→
   218→  const arabe = langue === 'ar'
   219→  /* Vue du lecteur : le TEXTE RÉÉCRIT d'abord — encre sur porcelaine,
   220→     polices et couleurs réglables, évolutif. La page numérisée reste
   221→     disponible pour lire le livre imprimé à l'identique. La recherche ⌘K
   222→     ouvre directement le texte au chapitre demandé. */
   223→  const aLesDeux = Boolean(pdf) && chapitres.length > 0
   224→  const [vue, setVue] = useState<'doc' | 'texte'>(
   225→    chapitreInitial || !pdf || chapitres.length > 0 || estSama ? 'texte' : 'doc'
   226→  )
   227→  const vueDoc = vue === 'doc' && Boolean(pdf)
   228→  const theme = THEMES.find((t) => t.id === l.theme) ?? THEMES[0]
   229→  const accent = ACCENTS.find((a) => a.id === l.accent) ?? ACCENTS[0]
   230→  const police = arabe
   231→    ? POLICES_AR.find((p) => p.id === l.policeArabe) ?? POLICES_AR[0]
   232→    : POLICES_PH.find((p) => p.id === l.police) ?? POLICES_PH[0]
   233→  const taille = arabe ? l.tailleArabe : l.taille
   234→
   235→  /* ─── L'écran ne s'éteint pas pendant qu'on lit ───
   236→     Un frère qui récite ne touche pas son téléphone pendant de
   237→     longues minutes. Le verrou d'écran est demandé à l'ouverture,
   238→     rendu si la page passe à l'arrière-plan, et relâché après un
   239→     quart d'heure sans le moindre geste — un téléphone oublié
   240→     écran allumé chauffe. Tout échec est muet. */
   241→  useEffect(() => {
   242→    if (!('wakeLock' in navigator)) return
   243→    let verrou: WakeLockSentinel | null = null
   244→    let minuteur: number | undefined
   245→    const REPOS = 15 * 60 * 1000
   246→
   247→    const demander = async () => {
   248→      try {
   249→        verrou = await navigator.wakeLock.request('screen')
   250→      } catch {
   251→        /* appareil qui ne sait pas faire : on lit comme avant */
   252→      }
   253→    }
   254→    const relacher = () => {
   255→      verrou?.release().catch(() => undefined)
   256→      verrou = null
   257→    }
   258→    const toucher = () => {
   259→      window.clearTimeout(minuteur)
   260→      minuteur = window.setTimeout(relacher, REPOS)
   261→    }
   262→    const surVisibilite = () => {
   263→      if (document.visibilityState === 'visible') {
   264→        demander()
   265→        toucher()
   266→      } else {
   267→        relacher()
   268→      }
   269→    }
   270→    demander()
   271→    toucher()
   272→    document.addEventListener('visibilitychange', surVisibilite)
   273→    window.addEventListener('pointerdown', toucher, { passive: true })
   274→    return () => {
   275→      relacher()
   276→      window.clearTimeout(minuteur)
   277→      document.removeEventListener('visibilitychange', surVisibilite)
   278→      window.removeEventListener('pointerdown', toucher)
   279→    }
   280→  }, [])
   281→
   282→  /* ─── Le corps de la page ne défile pas derrière le lecteur ─── */
   283→  useEffect(() => {
   284→    const deplacement = window.scrollY
   285→    document.body.style.overflow = 'hidden'
   286→    return () => {
   287→      document.body.style.overflow = ''
   288→      window.scrollTo(0, deplacement)
   289→    }
   290→  }, [])
   291→
   292→  /* ─── Échap ferme le lecteur ─── */
   293→  useEffect(() => {
   294→    const surTouche = (e: KeyboardEvent) => {
   295→      if (e.key === 'Escape') onFermer()
   296→    }
   297→    window.addEventListener('keydown', surTouche)
   298→    return () => window.removeEventListener('keydown', surTouche)
   299→  }, [onFermer])
   300→
   301→  /* ─── Progression de la lecture + affichage du bouton revenir ─── */
   302→  useEffect(() => {
   303→    const el = conteneurRef.current
   304→    if (!el) return
   305→    const surScroll = () => {
   306→      const reste = el.scrollHeight - el.clientHeight
   307→      setProgression(reste > 0 ? Math.min(1, el.scrollTop / reste) : 0)
   308→      setScrolle(el.scrollTop > 260)
   309→    }
   310→    el.addEventListener('scroll', surScroll, { passive: true })
   311→    surScroll()
   312→    return () => el.removeEventListener('scroll', surScroll)
   313→  }, [vueDoc])
   314→
   315→  /* ─── Positionner au chapitre demandé à l'ouverture ───
   316→     (le diwân arrive un instant après : on replace alors)
   317→     « langue » n'est PAS une dépendance : basculer phonétique ⇄ عربي
   318→     ne doit jamais replacer la lecture — les pratiques unifiées
   319→     gardent leur place, seul le texte affiché change. */
   320→  useEffect(() => {
   321→    const el = conteneurRef.current
   322→    if (!el) return
   323→    if (chapitreInitial) {
   324→      const i = chapitres.findIndex((c) => c.id === chapitreInitial)
   325→      const cible = i >= 0 ? cartesRef.current[i] : null
   326→      if (cible) {
   327→        el.scrollTo({ top: cible.offsetTop - el.offsetTop - 8 })
   328→        return
   329→      }
   330→    }
   331→    el.scrollTo({ top: 0 })
   332→  }, [chapitreInitial, doc.id, chapitres.length])
   333→
   334→  const allerAuChapitre = useCallback(
   335→    (i: number) => {
   336→      const cible = cartesRef.current[i]
   337→      if (cible) {
   338→        conteneurRef.current?.scrollTo({
   339→          top: cible.offsetTop - (conteneurRef.current?.offsetTop ?? 0) - 8,
   340→          behavior: 'smooth',
   341→        })
   342→      }
   343→    },
   344→    []
   345→  )
   346→
   347→  /* ─── Le chapitre visible suit le défilement ─── */
   348→  useEffect(() => {
   349→    if (chapitres.length < 2) return
   350→    const el = conteneurRef.current
   351→    if (!el) return
   352→    const observateur = new IntersectionObserver(
   353→      (entrees) => {
   354→        for (const e of entrees) {
   355→          if (e.isIntersecting) {
   356→            const i = Number((e.target as HTMLElement).dataset.i)
   357→            if (!Number.isNaN(i)) setIndexActif(i)
   358→          }
   359→        }
   360→      },
   361→      { root: el, rootMargin: '-15% 0px -60% 0px', threshold: 0 }
   362→    )
   363→    cartesRef.current.forEach((c) => c && observateur.observe(c))
   364→    return () => observateur.disconnect()
   365→  }, [chapitres.length, langue, doc.id])
   366→
   367→  const titre = arabe ? doc.titreAr : doc.titreFr
   368→
   369→  return (
   370→    <motion.div
   371→      initial={{ opacity: 0 }}
   372→      animate={{ opacity: 1 }}
   373→      exit={{ opacity: 0 }}
   374→      transition={{ duration: 0.22 }}
   375→      /* mb-0 : parade contre les space-y-* d'un ancêtre (Tailwind v4
   376→          pose margin-block-end sur :not(:last-child)) — un lecteur plein
   377→          écran ne doit jamais recevoir de marge, où qu'il vive */
   378→      className="fixed inset-0 z-[90] mb-0 flex flex-col bg-background grain-papier"
   379→      role="dialog"
   380→      aria-modal="true"
   381→      aria-label={`Lecteur — ${titre}`}
   382→    >
   383→      {/* ─── La barre du haut : titre, sommaire, réglages, fermeture ─── */}
   384→      <header className="relative z-10 flex-none border-b border-border bg-card/95 backdrop-blur supports-[backdrop-filter]:bg-card/90">
   385→        <div
   386→          className="flex items-center gap-1 px-2.5"
   387→          style={{ paddingTop: 'calc(10px + env(safe-area-inset-top))', paddingBottom: 10 }}
   388→        >
   389→          {/* Réglages de lecture — le texte seulement */}
   390→          {!vueDoc && (
   391→            <button
   392→              type="button"
   393→              onClick={() => {
   394→                setReglagesOuverts((o) => !o)
   395→                setSommaireOuvert(false)
   396→              }}
   397→              aria-expanded={reglagesOuverts}
   398→              aria-label="Réglages de lecture"
   399→              className={cn(
   400→                'grid h-9 w-9 place-items-center rounded-full transition-colors duration-200',
   401→                reglagesOuverts
   402→                  ? 'bg-muted text-foreground'
   403→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   404→              )}
   405→            >
   406→              <Type className="h-4.5 w-4.5" />
   407→            </button>
   408→          )}
   409→
   410→          {/* Sommaire des chapitres */}
   411→          {!vueDoc && chapitres.length > 1 && (
   412→            <button
   413→              type="button"
   414→              onClick={() => {
   415→                if (doc.pratiques) {
   416→                  /* Le cahier de wadhifa : le sommaire est la carte en tête */
   417→                  setSommaireCarte((o) => !o)
   418→                  conteneurRef.current?.scrollTo({ top: 0, behavior: 'smooth' })
   419→                } else {
   420→                  setSommaireOuvert((o) => !o)
   421→                }
   422→                setReglagesOuverts(false)
   423→              }}
   424→              aria-expanded={sommaireOuvert || (doc.pratiques && sommaireCarte)}
   425→              aria-label="Sommaire du document"
   426→              className={cn(
   427→                'grid h-9 w-9 place-items-center rounded-full transition-colors duration-200',
   428→                sommaireOuvert || (doc.pratiques && sommaireCarte)
   429→                  ? 'bg-muted text-foreground'
   430→                  : 'text-brouillard hover:bg-muted/60 hover:text-foreground'
   431→              )}
   432→            >
   433→              <List className="h-4.5 w-4.5" />
   434→            </button>
   435→          )}
   436→
   437→          {/* Le titre, centré — le nom de la pièce */}
   438→          <h2
   439→            className={cn(
   440→              'min-w-0 flex-1 truncate text-center text-[16px] font-medium text-encre',
   441→              arabe && 'font-arabic text-[19px]'
   442→            )}
   443→            dir={arabe ? 'rtl' : undefined}
   444→            title={titre}
   445→          >
   446→            {titre}
   447→          </h2>
   448→
   449→          {/* Fermeture */}
   450→          <button
   451→            type="button"
   452→            onClick={onFermer}
   453→            aria-label="Fermer le lecteur"
   454→            className="grid h-9 w-9 place-items-center rounded-full border border-border text-brouillard transition-colors duration-200 hover:bg-muted/60 hover:text-foreground active:bg-muted"
   455→          >
   456→            <X className="h-4.5 w-4.5" />
   457→          </button>
   458→        </div>
   459→
   460→        {/* Filet de progression de la lecture */}
   461→        <div className="absolute inset-x-0 bottom-0 h-[2px] bg-border" aria-hidden>
   462→          <div
   463→            className="h-full bg-primary transition-[width] duration-150"

   468→        {/* Bascule Texte / Numérisé — le texte réécrit d'abord (encre

   470→        {aLesDeux && (
