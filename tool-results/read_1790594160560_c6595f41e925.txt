    89→/* ─── Une ligne du texte, répétitions « x 100 » / « × ١٠٠ » en badge ─── */
    90→function LigneInvocation({ ligne, accent }: { ligne: string; accent: string }) {
    91→  const m = ligne.match(/^(.*?)(\s*[×x]\s*[٠-٩\dHh '’]+)$/u)
    92→  if (m && m[2].trim()) {
    93→    return (
    94→      <span className="block">
    95→        {m[1]}
    96→        <span
    97→          className="mx-1 inline-block rounded-md px-2 py-0.5 align-middle text-[0.72em] font-bold tracking-wide"
    98→          style={{ color: accent, backgroundColor: `${accent}18` }}
    99→        >
   100→          {m[2].trim()}
   101→        </span>
   102→      </span>
   103→    )
   104→  }
   105→  return <span className="block">{ligne || '\u00A0'}</span>
   106→}

   184→/* ─── Un vers de qaṣīda — deux hémistiches pleine largeur ───

   188→function VersPoeme({ ligne, accent }: { ligne: string; accent: string }) {
   189→  const coupe = ligne.split('۞')
   190→  if (coupe.length !== 2) {
   191→    return <LigneInvocation ligne={ligne} accent={accent} />
   192→  }
   193→  return (
   194→    <span className="poesie-vers">
   195→      <span className="poesie-shatr">{coupe[0].trim()}</span>
   196→      <span aria-hidden className="poesie-fasl" style={{ color: accent }}>
   197→        ۞
   198→      </span>
   199→      <span className="poesie-shatr">{coupe[1].trim()}</span>
   200→    </span>
   201→  )
   202→}

   204→/* ─── Le paragraphage du cahier phonétique ───
   205→   Le cahier imprimé est une colonne étroite : l'extraction garde
   206→   des lignes courtes coupées au milieu des phrases. On rejoint
   207→   les lignes d'un même bloc (séparé par une ligne vide) et on
   208→   coupe après les renvois « → p. » et les répétitions « x 100 » —
   209→   le texte coule alors sur toute la largeur, justifié comme un
   210→   livre, au lieu de rester collé à gauche. */
   211→function paragrapher(texte: string): string[] {
   212→  const paras: string[] = []
   213→  let courant: string[] = []
   214→  const flush = () => {
   215→    if (courant.length) {
   216→      paras.push(courant.join(' ').replace(/\s+/g, ' ').trim())
   217→      courant = []
   218→    }
   219→  }
   220→  for (const brute of texte.split('\n')) {
   221→    const t = brute.trim()
   222→    if (!t) {
   223→      flush()
   224→      continue
   225→    }
   226→    courant.push(t)
   227→    /* Fin logique de paragraphe : renvoi de page, ou répétition badgée */
   228→    if (/→\s*p\.|→\s*ص|[×x]\s*[٠-٩\d][٠-٩\dHh '’]*$/.test(t)) flush()
   229→  }
   230→  flush()
   231→  return paras
   232→}

   253→function LecteurAdhkar({ doc, langue, chapitreInitial, onFermer }: LecteurProps) {
   254→  const l = useLecture()
   255→  const imprimer = usePrint((s) => s.imprimer)
   256→
   257→  /* Le diwân de samâ‘ (398 chants) n'entre pas dans le fil principal :
   258→     il se charge paresseusement à l'ouverture du document. */
   259→  const estSama = doc.id === 'sama' && langue === 'ph'
   260→  const [samaTexte, setSamaTexte] = useState<ChapitreLecture[] | null>(null)
   261→  useEffect(() => {
   262→    if (doc.id !== 'sama' || langue !== 'ph') return
   263→    let annule = false
   264→    import('@/data/sama-chants').then((m) => {
   265→      if (!annule) setSamaTexte(m.CHAPITRES_SAMA)
   266→    })
   267→    return () => {
   268→      annule = true
   269→    }
   270→  }, [doc.id, langue])
   271→
   272→  /* Les entrées du lecteur : pratiques structurées (cahier de
   273→     wadhifa) ou chapitres plats (autres documents). Chaque entrée
   274→     porte sa langue effective — le sommaire ne change jamais quand
   275→     on bascule phonétique ⇄ عربي, la pratique lue reste en place. */
   276→  const entrees = useMemo<EntreeLecture[]>(() => {
   277→    if (doc.pratiques) {
   278→      return doc.pratiques
   279→        .map((p): EntreeLecture | null => {
   280→          const prefere = p[langue]
   281→          const autre = langue === 'ph' ? p.ar : p.ph
   282→          const c = prefere ?? autre
   283→          if (!c) return null
   284→          return {
   285→            ...c,
   286→            langue: prefere ? langue : langue === 'ph' ? 'ar' : 'ph',
   287→            repli: !prefere,
   288→            groupe: p.groupe,
   289→          }
   290→        })
   291→        .filter((e): e is EntreeLecture => e !== null)
   292→    }
   293→    if (doc.id === 'sama' && langue === 'ph') {
   294→      return (samaTexte ?? []).map((c) => ({ ...c, langue: 'ph' as const }))
   295→    }
   296→    return chapitresDuDoc(doc, langue).map((c) => ({ ...c, langue }))
   297→  }, [doc, langue, samaTexte])
   298→  const chapitres = entrees
   299→  const pdf = pdfDuDoc(doc, langue)
   300→  const conteneurRef = useRef<HTMLDivElement | null>(null)
   301→  const cartesRef = useRef<(HTMLElement | null)[]>([])
   302→
   303→  const [indexActif, setIndexActif] = useState(() => {
   304→    const i = chapitreInitial ? entrees.findIndex((c) => c.id === chapitreInitial) : -1
   305→    return i >= 0 ? i : 0
   306→  })
   307→  const [reglagesOuverts, setReglagesOuverts] = useState(false)
   308→  const [sommaireOuvert, setSommaireOuvert] = useState(false)
   309→  /* La carte sommaire du cahier — dépliée à l'ouverture, refermée
   310→     si la recherche ⌘K demande directement une pratique. */
   311→  const [sommaireCarte, setSommaireCarte] = useState(!chapitreInitial)
   312→  const [progression, setProgression] = useState(0)
   313→  const [scrolle, setScrolle] = useState(false)
   314→
   315→  const arabe = langue === 'ar'
   316→  /* Vue du lecteur : le TEXTE RÉÉCRIT d'abord — encre sur porcelaine,
   317→     polices et couleurs réglables, évolutif. La page numérisée reste
   318→     disponible pour lire le livre imprimé à l'identique. La recherche ⌘K
   319→     ouvre directement le texte au chapitre demandé. */
   320→  const aLesDeux = Boolean(pdf) && chapitres.length > 0
   321→  const [vue, setVue] = useState<'doc' | 'texte'>(
   322→    chapitreInitial || !pdf || chapitres.length > 0 || estSama ? 'texte' : 'doc'
   323→  )
   324→  const vueDoc = vue === 'doc' && Boolean(pdf)
   325→  const theme = THEMES.find((t) => t.id === l.theme) ?? THEMES[0]
   326→  const accent = ACCENTS.find((a) => a.id === l.accent) ?? ACCENTS[0]
   327→  const police = arabe
   328→    ? POLICES_AR.find((p) => p.id === l.policeArabe) ?? POLICES_AR[0]
   329→    : POLICES_PH.find((p) => p.id === l.police) ?? POLICES_PH[0]
   330→  const taille = arabe ? l.tailleArabe : l.taille
   331→
   332→  /* ─── L'écran ne s'éteint pas pendant qu'on lit ───
   333→     Un frère qui récite ne touche pas son téléphone pendant de
   334→     longues minutes. Le verrou d'écran est demandé à l'ouverture,
   335→     rendu si la page passe à l'arrière-plan, et relâché après un
   336→     quart d'heure sans le moindre geste — un téléphone oublié
   337→     écran allumé chauffe. Tout échec est muet. */
   338→  useEffect(() => {
   339→    if (!('wakeLock' in navigator)) return
   340→    let verrou: WakeLockSentinel | null = null
   341→    let minuteur: number | undefined
   342→    const REPOS = 15 * 60 * 1000
   343→
   344→    const demander = async () => {
   345→      try {
   346→        verrou = await navigator.wakeLock.request('screen')
   347→      } catch {
   348→        /* appareil qui ne sait pas faire : on lit comme avant */
   349→      }
   350→    }
   351→    const relacher = () => {
   352→      verrou?.release().catch(() => undefined)
   353→      verrou = null
   354→    }
   355→    const toucher = () => {
   356→      window.clearTimeout(minuteur)
   357→      minuteur = window.setTimeout(relacher, REPOS)
   358→    }
   359→    const surVisibilite = () => {
   360→      if (document.visibilityState === 'visible') {
   361→        demander()
   362→        toucher()
   363→      } else {
   364→        relacher()
   365→      }
   366→    }
   367→    demander()
   368→    toucher()
   369→    document.addEventListener('visibilitychange', surVisibilite)
   370→    window.addEventListener('pointerdown', toucher, { passive: true })
   371→    return () => {
   372→      relacher()
   373→      window.clearTimeout(minuteur)
   374→      document.removeEventListener('visibilitychange', surVisibilite)
   375→      window.removeEventListener('pointerdown', toucher)
   376→    }
   377→  }, [])
   378→
   379→  /* ─── Le corps de la page ne défile pas derrière le lecteur ─── */
   380→  useEffect(() => {
   381→    const deplacement = window.scrollY
   382→    document.body.style.overflow = 'hidden'
   383→    return () => {
   384→      document.body.style.overflow = ''
   385→      window.scrollTo(0, deplacement)
   386→    }
   387→  }, [])
   388→
   389→  /* ─── Échap ferme le lecteur ─── */
   390→  useEffect(() => {
   391→    const surTouche = (e: KeyboardEvent) => {
   392→      if (e.key === 'Escape') onFermer()
   393→    }
   394→    window.addEventListener('keydown', surTouche)
   395→    return () => window.removeEventListener('keydown', surTouche)
   396→  }, [onFermer])
   397→
   398→  /* ─── Progression de la lecture + affichage du bouton revenir ─── */
   399→  useEffect(() => {
   400→    const el = conteneurRef.current
   401→    if (!el) return
   402→    const surScroll = () => {
   403→      const reste = el.scrollHeight - el.clientHeight
   404→      setProgression(reste > 0 ? Math.min(1, el.scrollTop / reste) : 0)
   405→      setScrolle(el.scrollTop > 260)
   406→    }
   407→    el.addEventListener('scroll', surScroll, { passive: true })
   408→    surScroll()
   409→    return () => el.removeEventListener('scroll', surScroll)
   410→  }, [vueDoc])
   411→
   412→  /* ─── Positionner au chapitre demandé à l'ouverture ───
   413→     (le diwân arrive un instant après : on replace alors)
   414→     « langue » n'est PAS une dépendance : basculer phonétique ⇄ عربي
   415→     ne doit jamais replacer la lecture — les pratiques unifiées
   416→     gardent leur place, seul le texte affiché change. */
   417→  useEffect(() => {
   418→    const el = conteneurRef.current
   419→    if (!el) return
   420→    if (chapitreInitial) {
   421→      const i = chapitres.findIndex((c) => c.id === chapitreInitial)
   422→      const cible = i >= 0 ? cartesRef.current[i] : null
   423→      if (cible) {
   424→        cible.scrollIntoView({ block: 'start' })
   425→        return
   426→      }
   427→    }
   428→    el.scrollTo({ top: 0 })
   429→  }, [chapitreInitial, doc.id, chapitres.length])
   430→
   431→  /* Ancre robuste : scrollIntoView + marge de défilement — le
   432→     navigateur mesure lui-même la position réelle, quel que soit
   433→     l'état de l'en-tête ou des panneaux déroulés au-dessus. */
   434→  const allerAuChapitre = useCallback((i: number) => {
   435→    const cible = cartesRef.current[i]
   436→    if (cible) cible.scrollIntoView({ behavior: 'smooth', block: 'start' })
   437→  }, [])
   438→
   439→  /* ─── Le chapitre visible suit le défilement ─── */
   440→  useEffect(() => {
   441→    if (chapitres.length < 2) return
   442→    const el = conteneurRef.current
   443→    if (!el) return
   444→    const observateur = new IntersectionObserver(
   445→      (entrees) => {
   446→        for (const e of entrees) {
   447→          if (e.isIntersecting) {
   448→            const i = Number((e.target as HTMLElement).dataset.i)
   449→            if (!Number.isNaN(i)) setIndexActif(i)
   450→          }
   451→        }
   452→      },
   453→      { root: el, rootMargin: '-15% 0px -60% 0px', threshold: 0 }
   454→    )
   455→    cartesRef.current.forEach((c) => c && observateur.observe(c))
   456→    return () => observateur.disconnect()
   457→  }, [chapitres.length, langue, doc.id])
   458→
   459→  const titre = arabe ? doc.titreAr : doc.titreFr
   460→
   461→  return (
   462→    <motion.div
   463→      initial={{ opacity: 0 }}
   464→      animate={{ opacity: 1 }}
   465→      exit={{ opacity: 0 }}
   466→      transition={{ duration: 0.22 }}
   467→      /* mb-0 : parade contre les space-y-* d'un ancêtre (Tailwind v4
   468→          pose margin-block-end sur :not(:last-child)) — un lecteur plein
   469→          écran ne doit jamais recevoir de marge, où qu'il vive */
   470→      className="fixed inset-0 z-[90] mb-0 flex flex-col bg-background grain-papier"
   471→      role="dialog"
   472→      aria-modal="true"
   473→      aria-label={`Lecteur — ${titre}`}
   474→    >
   475→      {/* ─── La barre du haut : titre, sommaire, réglages, fermeture ─── */}
   476→      <header className="relative z-10 flex-none border-b border-border bg-card/95 backdrop-blur supports-[backdrop-filter]:bg-card/90">
