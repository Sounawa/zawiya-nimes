    12→/** PRNG arithmétique pur (mulberry32) — strictement identique serveur/client,
    13→ *  contrairement à Math.sin dont la précision varie selon le moteur JS. */
    14→function mulberry32(graine: number) {
    15→  return () => {
    16→    graine |= 0
    17→    graine = (graine + 0x6d2b79f5) | 0
    18→    let t = Math.imul(graine ^ (graine >>> 15), 1 | graine)
    19→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    20→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    21→  }
    22→}
