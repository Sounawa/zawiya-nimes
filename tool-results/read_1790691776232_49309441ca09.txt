    88→/* ─── Une ligne du texte, répétitions « x 100 » / « × ١٠٠ » en badge ─── */
    89→function LigneInvocation({ ligne, accent }: { ligne: string; accent: string }) {
    90→  const m = ligne.match(/^(.*?)(\s*[×x]\s*[٠-٩\dHh '’]+)$/u)
    91→  if (m && m[2].trim()) {
    92→    return (
    93→      <span className="block">
    94→        {m[1]}
    95→        <span
    96→          className="mx-1 inline-block rounded-md px-2 py-0.5 align-middle text-[0.72em] font-bold tracking-wide"
    97→          style={{ color: accent, backgroundColor: `${accent}18` }}
    98→        >
    99→          {m[2].trim()}
   100→        </span>
   101→      </span>
   102→    )
   103→  }
   104→  return <span className="block">{ligne || '\u00A0'}</span>
   105→}

   219→/* Le vers à deux hémistiches — monté seulement quand le ۞ sépare */
   220→function VersDeuxShatr({
   221→  shatrDroite,
   222→  shatrGauche,
   223→  accent,
   224→  taille,
   225→  police,
   226→}: {
   227→  shatrDroite: string
   228→  shatrGauche: string
   229→  accent: string
   230→  taille: number
   231→  police: string
   232→}) {
   233→  const ref = useRef<HTMLSpanElement | null>(null)
   234→
   235→  useLayoutEffect(() => {
   236→    const el = ref.current
   237→    if (!el) return
   238→
   239→    const ajuster = () => {
   240→      /* D'abord rendre la taille choisie par le frère */
   241→      el.style.fontSize = ''
   242→      /* Hors du mode une-ligne (tablette, paysage : deux colonnes
   243→         du kitāb), la taille choisie s'applique telle quelle */
   244→      if (getComputedStyle(el).display !== 'flex') return
   245→      const shatrs = el.querySelectorAll('.poesie-shatr')
   246→      const fasl = el.querySelector('.poesie-fasl')
   247→      if (shatrs.length !== 2 || !fasl) return
   248→
   249→      /* Largeur naturelle du vers entier, nowrap */
   250→      const gap =
   251→        2 * (parseFloat(getComputedStyle(el).columnGap) || 0) || 0
   252→      const nat =
   253→        shatrs[0].getBoundingClientRect().width +
   254→        shatrs[1].getBoundingClientRect().width +
   255→        fasl.getBoundingClientRect().width +
   256→        gap
   257→      const dispo = el.clientWidth
   258→      if (nat <= dispo + 1 || nat === 0) return
   259→
   260→      /* Resserrer juste ce qu'il faut — jamais au-dessous de 11 px.
   261→         Deux pixels de grâce : les largeurs mesurées sont arrondies
   262→         au quart de pixel près par le moteur de rendu. */
   263→      const heritee = parseFloat(getComputedStyle(el).fontSize)
   264→      el.style.fontSize = `${Math.max(11, Math.floor((heritee * (dispo - 2)) / nat))}px`
   265→    }
   266→
   267→    ajuster()
   268→    /* La fenêtre bouge (rotation, redimensionnement) : re-mesurer */
   269→    const ro = new ResizeObserver(ajuster)
   270→    ro.observe(el)
   271→    return () => ro.disconnect()
   272→  }, [shatrDroite, shatrGauche, taille, police])
   273→
   274→  return (
   275→    <span ref={ref} className="poesie-vers">
   276→      <span className="poesie-shatr">{shatrDroite}</span>
   277→      <span aria-hidden className="poesie-fasl" style={{ color: accent }}>
   278→        ۞
   279→      </span>
   280→      <span className="poesie-shatr">{shatrGauche}</span>
   281→    </span>
   282→  )
   283→}

   285→/* ─── Le paragraphage du cahier phonétique ───
   286→   Le cahier imprimé est une colonne étroite : l'extraction garde
   287→   des lignes courtes coupées au milieu des phrases. On rejoint
   288→   les lignes d'un même bloc (séparé par une ligne vide) et on
   289→   coupe après les renvois « → p. » et les répétitions « x 100 » —
   290→   le texte coule alors sur toute la largeur, justifié comme un
   291→   livre, au lieu de rester collé à gauche. */
   292→function paragrapher(texte: string): string[] {
   293→  const paras: string[] = []
   294→  let courant: string[] = []
   295→  const flush = () => {
   296→    if (courant.length) {
   297→      paras.push(courant.join(' ').replace(/\s+/g, ' ').trim())
   298→      courant = []
   299→    }
   300→  }
   301→  for (const brute of texte.split('\n')) {
   302→    const t = brute.trim()
   303→    if (!t) {
   304→      flush()
   305→      continue
   306→    }
   307→    courant.push(t)
   308→    /* Fin logique de paragraphe : renvoi de page, ou répétition badgée */
   309→    if (/→\s*p\.|→\s*ص|[×x]\s*[٠-٩\d][٠-٩\dHh '’]*$/.test(t)) flush()
   310→  }
   311→  flush()
   312→  return paras
   313→}

   334→function LecteurAdhkar({ doc, langue, chapitreInitial, onFermer }: LecteurProps) {
   335→  const l = useLecture()
   336→  const imprimer = usePrint((s) => s.imprimer)

   338→  /* Le diwân de samâ‘ (398 chants) n'entre pas dans le fil principal :
   339→     il se charge paresseusement à l'ouverture du document. */
   340→  const estSama = doc.id === 'sama' && langue === 'ph'
   341→  const [samaTexte, setSamaTexte] = useState<ChapitreLecture[] | null>(null)
   342→  useEffect(() => {
   343→    if (doc.id !== 'sama' || langue !== 'ph') return
   344→    let annule = false
   345→    import('@/data/sama-chants').then((m) => {
   346→      if (!annule) setSamaTexte(m.CHAPITRES_SAMA)
   347→    })

   350→    }
   351→  }, [doc.id, langue])
   352→

   354→     wadhifa) ou chapitres plats (autres documents). Chaque entrée
   355→     porte sa langue effective — le sommaire ne change jamais quand
   356→     on bascule phonétique ⇄ عربي, la pratique lue reste en place.
   357→     ORDRE : en phonétique, l'ordre du cahier imprimé ; en arabe,
   358→     l'ordre du kitāb imprimé (ce qui est ensemble dans le livre
   359→     reste ensemble, dans l'ordre — Yā-Sīn p.4 d'abord, Ṣalāt
   360→     'alâ an-Nabī p.65 en dernier). Les pièces sans chapitre arabe
   361→     (absentes du kitāb) suivent, dans l'ordre du cahier. */
   362→  const entrees = useMemo<EntreeLecture[]>(() => {
   363→    if (doc.pratiques) {
   364→      const converties = doc.pratiques
   365→        .map((p): EntreeLecture | null => {
   366→          const prefere = p[langue]
   367→          const autre = langue === 'ph' ? p.ar : p.ph
   368→          const c = prefere ?? autre
   369→          if (!c) return null
   370→          return {
   371→            ...c,
   372→            langue: prefere ? langue : langue === 'ph' ? 'ar' : 'ph',
   373→            repli: !prefere,
   374→            groupe: p.groupe,
   375→          }
   376→        })
   377→        .filter((e): e is EntreeLecture => e !== null)
   378→      if (langue !== 'ar') return converties
   379→      const positionKitab = (id: string) => ORDRE_KITAB.indexOf(id)
   380→      return converties.sort((a, b) => {
   381→        const pa = positionKitab(a.id)
   382→        const pb = positionKitab(b.id)
   383→        if (pa >= 0 && pb >= 0) return pa - pb
   384→        if (pa >= 0) return -1
   385→        if (pb >= 0) return 1
   386→        return 0
   387→      })
   388→    }
   389→    if (doc.id === 'sama' && langue === 'ph') {
   390→      return (samaTexte ?? []).map((c) => ({ ...c, langue: 'ph' as const }))
   391→    }
   392→    return chapitresDuDoc(doc, langue).map((c) => ({ ...c, langue }))
   393→  }, [doc, langue, samaTexte])
   394→  const chapitres = entrees
   395→  const pdf = pdfDuDoc(doc, langue)
   396→  const conteneurRef = useRef<HTMLDivElement | null>(null)
   397→  const cartesRef = useRef<(HTMLElement | null)[]>([])
   398→
   399→  const [indexActif, setIndexActif] = useState(() => {
   400→    const i = chapitreInitial ? entrees.findIndex((c) => c.id === chapitreInitial) : -1
   401→    return i >= 0 ? i : 0
   402→  })
   403→  const [reglagesOuverts, setReglagesOuverts] = useState(false)
   404→  const [sommaireOuvert, setSommaireOuvert] = useState(false)
   405→  /* La carte sommaire du cahier — dépliée à l'ouverture, refermée
   406→     si la recherche ⌘K demande directement une pratique. */
   407→  const [sommaireCarte, setSommaireCarte] = useState(!chapitreInitial)
   408→  const [progression, setProgression] = useState(0)
   409→
   410→  const arabe = langue === 'ar'
   411→  /* Vue du lecteur : le TEXTE RÉÉCRIT d'abord — encre sur porcelaine,
   412→     polices et couleurs réglables, évolutif. La page numérisée reste
   413→     disponible pour lire le livre imprimé à l'identique. La recherche ⌘K
   414→     ouvre directement le texte au chapitre demandé. */
   415→  const aLesDeux = Boolean(pdf) && chapitres.length > 0
   416→  const [vue, setVue] = useState<'doc' | 'texte'>(
   417→    chapitreInitial || !pdf || chapitres.length > 0 || estSama ? 'texte' : 'doc'
   418→  )
   419→  const vueDoc = vue === 'doc' && Boolean(pdf)
   420→  const theme = THEMES.find((t) => t.id === l.theme) ?? THEMES[0]
   421→  const accent = ACCENTS.find((a) => a.id === l.accent) ?? ACCENTS[0]
   422→  const police = arabe
   423→    ? POLICES_AR.find((p) => p.id === l.policeArabe) ?? POLICES_AR[0]
   424→    : POLICES_PH.find((p) => p.id === l.police) ?? POLICES_PH[0]
   425→  const taille = arabe ? l.tailleArabe : l.taille
   426→
   427→  /* ─── L'écran ne s'éteint pas pendant qu'on lit ───
   428→     Un frère qui récite ne touche pas son téléphone pendant de
   429→     longues minutes. Le verrou d'écran est demandé à l'ouverture,
   430→     rendu si la page passe à l'arrière-plan, et relâché après un
   431→     quart d'heure sans le moindre geste — un téléphone oublié
   432→     écran allumé chauffe. Tout échec est muet. */
   433→  useEffect(() => {
   434→    if (!('wakeLock' in navigator)) return
   435→    let verrou: WakeLockSentinel | null = null
   436→    let minuteur: number | undefined
   437→    const REPOS = 15 * 60 * 1000
   438→
   439→    const demander = async () => {
   440→      try {
   441→        verrou = await navigator.wakeLock.request('screen')
   442→      } catch {
   443→        /* appareil qui ne sait pas faire : on lit comme avant */
   444→      }
   445→    }
   446→    const relacher = () => {
   447→      verrou?.release().catch(() => undefined)
   448→      verrou = null
   449→    }
   450→    const toucher = () => {
   451→      window.clearTimeout(minuteur)
   452→      minuteur = window.setTimeout(relacher, REPOS)
   453→    }
   454→    const surVisibilite = () => {
   455→      if (document.visibilityState === 'visible') {
   456→        demander()
   457→        toucher()
   458→      } else {
   459→        relacher()
   460→      }
   461→    }
   462→    demander()
   463→    toucher()
   464→    document.addEventListener('visibilitychange', surVisibilite)
   465→    window.addEventListener('pointerdown', toucher, { passive: true })
   466→    return () => {
   467→      relacher()
   468→      window.clearTimeout(minuteur)
   469→      document.removeEventListener('visibilitychange', surVisibilite)
   470→      window.removeEventListener('pointerdown', toucher)
   471→    }
   472→  }, [])
   473→
   474→  /* ─── Le corps de la page ne défile pas derrière le lecteur ─── */
   475→  useEffect(() => {
   476→    const deplacement = window.scrollY
   477→    document.body.style.overflow = 'hidden'
   478→    return () => {
   479→      document.body.style.overflow = ''
   480→      window.scrollTo(0, deplacement)
   481→    }
   482→  }, [])
   483→
