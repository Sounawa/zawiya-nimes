     5→import {
     6→  BadgeCheck,

     9→  Check,

    33→  type FamilleDroits,
    34→  type Member,
    35→  type Surcharges,
    36→  type Ville,
    37→} from '@/lib/types'
    38→import {
    39→  Dialog,
    40→  DialogContent,
    41→  DialogDescription,
    42→  DialogFooter,
    43→  DialogHeader,
    44→  DialogTitle,
    45→} from '@/components/ui/dialog'
    46→import { Button } from '@/components/ui/button'
    47→import { Input } from '@/components/ui/input'
    48→import { Label } from '@/components/ui/label'
    49→import { Switch } from '@/components/ui/switch'

    53→const JOURS = ['Lundi', 'Mardi', 'Mercredi', 'Jeudi', 'Vendredi', 'Samedi', 'Dimanche']
    54→
    55→/* Cycle des zawiya : Nîmes → Avignon → Montpellier → Nîmes */
    56→const CYCLE_VILLE: Record<string, Ville> = {
    57→  NIMES: 'AVIGNON',
    58→  AVIGNON: 'MONTPELLIER',
    59→  MONTPELLIER: 'NIMES',
    60→}
    61→
    62→/** Le droit du groupe, sans surcharge — pour distinguer défaut et particulier */
    63→const DROIT_DE_LA_FAMILLE: Record<FamilleDroits, Parameters<typeof peut>[2]> = {
    64→  annonces: 'annonces.creer',
    65→  sondages: 'sondages.creer',
    66→  reunions: 'reunions.rediger',
    67→  contenus: 'contenus.editer',
    68→  cotisations: 'cotisations.voir',
    69→}

    71→/* ═══════════════ Réglages : jours d'ijtima + frères + accès ═══════════════ */
    72→export function SettingsDialog({
    73→  open,
    74→  onOpenChange,
    75→}: {
    76→  open: boolean
    77→  onOpenChange: (v: boolean) => void
    78→}) {
    79→  const { data: settings } = useSettings()
    80→  const { data: membres } = useMembers()
    81→  const saveSetting = useSaveSetting()
    82→  const qc = useQueryClient()
    83→  const { toast } = useToast()
    84→
    85→  const joursDefaut: { label: string; actif: boolean; heure: string }[] =
    86→    settings?.ijtima_days?.jours ?? []
    87→  const [joursEdites, setJoursEdites] = useState<{
    88→    label: string
    89→    actif: boolean
    90→    heure: string
    91→  }[] | null>(null)
    92→  const jours = joursEdites ?? joursDefaut
    93→  const setJours = setJoursEdites
    94→  const [nouveauNom, setNouveauNom] = useState('')
    95→  const [villeAjout, setVilleAjout] = useState<Ville>('NIMES')
    96→  const [membreDroits, setMembreDroits] = useState<Member | null>(null)
    97→
    98→  /* ─── Lieu des ijtima' ─── */
    99→  const configDefaut = settings?.app_config ?? { nom: 'Nimes', ville: 'Nîmes', lieu: '' }
   100→  const [lieuEdite, setLieuEdite] = useState<{ ville: string; adresse: string } | null>(null)
   101→  const lieu = lieuEdite ?? { ville: configDefaut.ville, adresse: configDefaut.lieu ?? '' }
   102→
   103→  const enregistrerLieu = () => {
   104→    saveSetting.mutate(
   105→      {
   106→        key: 'app_config',
   107→        value: {
   108→          nom: configDefaut.nom || 'Nimes',
   109→          ville: lieu.ville.trim() || 'Nîmes',
   110→          lieu: lieu.adresse.trim(),
   111→        },
   112→      },
   113→      {
   114→        onSuccess: () =>
   115→          toast({
   116→            title: 'Lieu enregistré',
   117→            description: 'Les nouvelles réunions utiliseront ce lieu par défaut.',
   118→          }),
   119→      }
   120→    )
   121→  }
   122→
   123→  const majJour = (i: number, patch: Partial<{ actif: boolean; heure: string }>) =>
   124→    setJours((js) => (js ?? joursDefaut).map((j, k) => (k === i ? { ...j, ...patch } : j)))
   125→
   126→  const ajouterJour = () => {
   127→    const absent = JOURS.find((j) => !jours.some((x) => x.label === j))
   128→    if (absent) setJours((js) => [...(js ?? joursDefaut), { label: absent, actif: true, heure: '19:30' }])
   129→  }
   130→
   131→  const retirerJour = (i: number) =>
   132→    setJours((js) => (js ?? joursDefaut).filter((_, k) => k !== i))
   133→
   134→  const enregistrerJours = () => {
   135→    saveSetting.mutate(
   136→      { key: 'ijtima_days', value: { jours } },
   137→      {
   138→        onSuccess: () =>
   139→          toast({
   140→            title: 'Jours enregistrés',
   141→            description: 'Les nouveaux ijtima\u2018 sont pris en compte.',
   142→          }),
   143→      }
   144→    )
   145→  }
   146→
   147→  /* ─── Membres : rôles, groupes, droits, accès ─── */
   148→  const majMembre = useMutation({
   149→    mutationFn: ({
   150→      id,
   151→      ...data
   152→    }: {
   153→      id: string
   154→      role?: string
   155→      city?: string
   156→      status?: string
   157→      perms?: Record<string, boolean> | null
   158→      resetPin?: boolean
   159→    }) => api(`/api/members/${id}`, { method: 'PATCH', body: JSON.stringify(data) }),
   160→    onSuccess: () => {
   161→      qc.invalidateQueries({ queryKey: ['members'] })
   162→    },
   163→  })
   164→
   165→  const retirerMembre = useMutation({
   166→    mutationFn: (id: string) =>
   167→      api(`/api/members/${id}`, { method: 'DELETE', body: JSON.stringify({}) }),
   168→    onSuccess: () => {
   169→      qc.invalidateQueries({ queryKey: ['members'] })
   170→    },
   171→  })
   172→
   173→  const ajouterMembre = useMutation({
   174→    mutationFn: (name: string) =>
   175→      api('/api/members', {
   176→        method: 'POST',
   177→        body: JSON.stringify({ name, ville: villeAjout, role: 'MEMBER' }),
   178→      }),
   179→    onSuccess: () => {
   180→      qc.invalidateQueries({ queryKey: ['members'] })
   181→      toast({
   182→        title: 'Membre ajouté',
   183→        description: `Sidi ${nouveauNom} a rejoint la zawiya de ${VILLE_LABEL[villeAjout]}. Il posera son code à son premier passage.`,
   184→      })
   185→      setNouveauNom('')
   186→    },
   187→    onError: () => toast({ title: 'Impossible d\u2019ajouter', variant: 'destructive' }),
   188→  })
   189→
   190→  const referent = (membres ?? []).find((m) => m.role === 'REFERENT')
   191→  const demandes = (membres ?? []).filter((m) => m.status === 'PENDING')
   192→  const actifs = (membres ?? []).filter((m) => m.status !== 'PENDING')
   193→
   194→  /* Basculer la zawiya d'un frère — Nîmes → Avignon → Montpellier */
   195→  const basculerVille = (m: Member) => {
   196→    const autre = CYCLE_VILLE[m.city] ?? 'NIMES'
   197→    majMembre.mutate({ id: m.id, city: autre })
   198→    toast({
   199→      title: `${m.name} → ${VILLE_LABEL[autre]}`,
   200→      description:
   201→        autre === 'NIMES'
   202→          ? 'Il retrouve tous les droits des frères de Nîmes.'
   203→          : `Il passe en lecture et participation, comme Avignon et Montpellier — ses droits particuliers restent.`,
   204→    })
   205→  }
   206→
   207→  return (
   208→    <Dialog open={open} onOpenChange={onOpenChange}>
   209→      <DialogContent className="max-h-[88vh] max-w-lg overflow-y-auto rounded-3xl border-primary/15 bg-ivoire p-6">
   210→        <DialogHeader>
   211→          <DialogTitle className="font-display text-2xl text-encre">
   212→            Réglages de la zaouia
   213→          </DialogTitle>
   214→          <DialogDescription>
   215→            Jours d'ijtima et conduite des frères — du ressort du référent.
   216→          </DialogDescription>
   217→        </DialogHeader>
   218→
   219→        {/* ─── Demandes d'inscription ─── */}
   220→        {demandes.length > 0 && (
   221→          <section aria-label="Demandes d'inscription" className="space-y-3">
   222→            <h3 className="flex items-center gap-2 text-[11px] font-bold uppercase tracking-[0.25em] text-gold-vif">
   223→              <UserRoundPlus className="h-4 w-4" /> Demandes d'inscription
   224→              <span className="rounded-full bg-accent/20 px-2 py-0.5 text-[10px] font-bold text-gold-vif">
   225→                {demandes.length}
   226→              </span>
   227→            </h3>
   228→            <p className="rounded-xl bg-or-pale px-3 py-2 text-xs leading-relaxed text-gold-vif">
   229→              Ces frères ont renseigné leur prénom et leur code à la porte d'entrée —
   230→              ouvre-leur l'accès, ou décline la demande.
   231→            </p>
   232→            <div className="defile-fin max-h-52 space-y-1.5 overflow-y-auto rounded-2xl border border-accent/25 bg-card p-2 shadow-sm">
   233→              {demandes.map((m) => (
   234→                <motion.div
   235→                  layout
   236→                  key={m.id}
   237→                  className="flex flex-wrap items-center justify-between gap-2 rounded-xl bg-or-pale/60 px-3 py-2.5"
   238→                >
   239→                  <span className="flex min-w-0 items-center gap-2 text-sm font-semibold text-secondary-foreground">
   240→                    <Hourglass className="h-3.5 w-3.5 shrink-0 text-gold-vif" />
   241→                    <span className="truncate">Sidi {m.name}</span>
   242→                    <span className="shrink-0 rounded-full bg-accent/15 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wider text-gold-vif">
   243→                      {VILLE_LABEL[m.city] ?? m.city}
   244→                    </span>
   245→                  </span>
   246→                  <span className="flex shrink-0 gap-1.5">
   247→                    <Button
   248→                      size="sm"
   249→                      onClick={() =>
   250→                        majMembre.mutate(
   251→                          { id: m.id, status: 'ACTIVE' },
   252→                          {
   253→                            onSuccess: () =>
   254→                              toast({
   255→                                title: `Accès ouvert à Sidi ${m.name}`,
   256→                                description: 'Il peut entrer avec son code personnel.',
   257→                              }),
   258→                          }
   259→                        )
   260→                      }
   261→                      className="h-8 gap-1 rounded-full bg-primary px-3 text-[11px] text-white hover:bg-emerald-deep"
   262→                    >
   263→                      <Check className="h-3.5 w-3.5" /> Valider
   264→                    </Button>
   265→                    <Button
   266→                      size="sm"
   267→                      variant="outline"
   268→                      onClick={() =>
   269→                        retirerMembre.mutate(m.id, {
   270→                          onSuccess: () =>
   271→                            toast({ title: `Demande de Sidi ${m.name} déclinée` }),
   272→                        })
   273→                      }
   274→                      className="h-8 gap-1 rounded-full border-destructive/30 px-3 text-[11px] text-destructive hover:bg-destructive/10 hover:text-destructive"
   275→                    >
   276→                      <X className="h-3.5 w-3.5" /> Décliner
   277→                    </Button>
   278→                  </span>
   279→                </motion.div>
   280→              ))}
   281→            </div>
   282→          </section>
   283→        )}
   284→
   285→        {/* ─── Lieu des ijtima' ─── */}
   286→        <section aria-label="Lieu des ijtima" className="space-y-3">
   287→          <h3 className="flex items-center gap-2 text-[11px] font-bold uppercase tracking-[0.25em] text-primary">
   288→            <MapPin className="h-4 w-4" /> Lieu des ijtima'
   289→          </h3>
   290→          <div className="space-y-3 rounded-2xl border border-primary/12 bg-card p-4 shadow-sm">
   291→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
   292→              <div className="space-y-1.5">
   293→                <Label className="text-xs text-muted-foreground">Ville</Label>
   294→                <Input
   295→                  value={lieu.ville}
   296→                  onChange={(e) => setLieuEdite({ ...lieu, ville: e.target.value })}
   297→                  placeholder="Nîmes"
   298→                  className="rounded-xl border-primary/25 bg-card"
   299→                />
   300→              </div>
   301→              <div className="space-y-1.5 sm:col-span-2">
   302→                <Label className="text-xs text-muted-foreground">Adresse</Label>
   303→                <Input
   304→                  value={lieu.adresse}
   305→                  onChange={(e) => setLieuEdite({ ...lieu, adresse: e.target.value })}
   306→                  placeholder="Rue, numéro, sonnette…"
   307→                  className="rounded-xl border-primary/25 bg-card"
   308→                />
   309→              </div>
   310→            </div>
   311→            <Button
   312→              size="sm"
   313→              onClick={enregistrerLieu}
   314→              disabled={saveSetting.isPending}
   315→              className="rounded-full bg-primary text-white hover:bg-emerald-deep"
   316→            >
   317→              Enregistrer le lieu
   318→            </Button>
   319→          </div>
   320→        </section>
   321→
   322→        {/* ─── Jours d'ijtima' ─── */}
   323→        <section aria-label="Jours d'ijtima" className="space-y-3">
   324→          <h3 className="flex items-center gap-2 text-[11px] font-bold uppercase tracking-[0.25em] text-primary">
   325→            <CalendarDays className="h-4 w-4" /> Jours des ijtima'
   326→          </h3>
   327→          <div className="rounded-2xl border border-primary/12 bg-card p-4 shadow-sm">
   328→            <ul className="space-y-2.5">
   329→              {jours.map((j, i) => (
   330→                <motion.li
   331→                  layout
   332→                  key={j.label}
   333→                  className="flex items-center gap-3 rounded-xl bg-background px-3 py-2"
   334→                >
   335→                  <Switch
   336→                    checked={j.actif}
   337→                    onCheckedChange={(v) => majJour(i, { actif: v })}
   338→                    aria-label={`${j.label} ${j.actif ? 'actif' : 'inactif'}`}
   339→                  />
   340→                  <span className={cn('flex-1 text-sm font-medium', j.actif ? 'text-secondary-foreground' : 'text-brouillard-clair line-through')}>
   341→                    {j.label}
   342→                  </span>
   343→                  <Input
   344→                    type="time"
   345→                    value={j.heure}
   346→                    onChange={(e) => majJour(i, { heure: e.target.value })}
   347→                    className="w-28 rounded-lg border-primary/25 bg-card text-sm"
   348→                  />
   349→                  <button
   350→                    onClick={() => retirerJour(i)}
   351→                    aria-label={`Retirer ${j.label}`}
   352→                    className="rounded-full p-2 text-brouillard-clair hover:bg-destructive/10 hover:text-destructive"
   353→                  >
   354→                    <Plus className="h-3.5 w-3.5 rotate-45" />
   355→                  </button>
   356→                </motion.li>
   357→              ))}
   358→            </ul>
   359→            <div className="mt-3 flex gap-2">
   360→              <Button
   361→                variant="outline"
   362→                size="sm"
   363→                onClick={ajouterJour}
   364→                className="gap-1.5 rounded-full border-dashed border-primary/40 text-primary"
   365→              >
   366→                <Plus className="h-3.5 w-3.5" /> Jour
   367→              </Button>
   368→              <Button
   369→                size="sm"
   370→                onClick={enregistrerJours}
   371→                disabled={saveSetting.isPending}
   372→                className="ml-auto rounded-full bg-primary text-white hover:bg-emerald-deep"
   373→              >
   374→                Enregistrer
   375→              </Button>
   376→            </div>
   377→          </div>
   378→        </section>
   379→
   380→        {/* ─── Membres : rôles, groupes, droits, accès ─── */}
   381→        <section aria-label="Membres" className="space-y-3">
   382→          <h3 className="flex items-center gap-2 text-[11px] font-bold uppercase tracking-[0.25em] text-primary">
   383→            <Shield className="h-4 w-4" /> Frères de la zaouia
   384→          </h3>
   385→          <p className="rounded-xl bg-sand/60 px-3 py-2 text-xs leading-relaxed text-muted-foreground">
   386→            Chaque frère porte un <strong>groupe</strong> (sa zawiya), un <strong>rôle</strong> et
   387→            d'éventuels <strong>droits particuliers</strong> : les frères de{' '}
   388→            <strong className="text-primary">Nîmes</strong> publient les annonces, sondages et
   389→            ijtima' ; ceux d'<strong className="text-gold-vif">Avignon</strong> et de{' '}
   390→            <strong className="text-gold-vif">Montpellier</strong> lisent et participent. Le bouton{' '}
   391→            <SlidersHorizontal className="inline h-3 w-3" /> accorde ou retire un droit précis à
   392→            n'importe quel frère.
   393→          </p>
   394→          <div className="defile-fin max-h-80 space-y-1 overflow-y-auto rounded-2xl border border-primary/12 bg-card p-2 shadow-sm">
   395→            {actifs.map((m) => (
   396→              <div
   397→                key={m.id}
   398→                className={cn(
   399→                  'flex flex-col gap-1.5 rounded-xl px-3 py-2 transition-colors hover:bg-background',
   400→                  m.status === 'BLOCKED' && 'opacity-75'
   401→                )}
   402→              >
   403→                <div className="flex flex-wrap items-center justify-between gap-x-2 gap-y-1.5">
   404→                  <span className="flex min-w-0 items-center gap-2 text-sm font-medium text-secondary-foreground">
   405→                    <UserRound className="h-3.5 w-3.5 shrink-0 text-primary/50" />
   406→                    <span className="truncate">Sidi {m.name}</span>
   407→                    <button
   408→                      onClick={() => basculerVille(m)}
   409→                      title={`Basculer vers ${VILLE_LABEL[CYCLE_VILLE[m.city] ?? 'NIMES']} — ${m.city === 'NIMES' ? 'il passera en lecture et participation' : 'il retrouvera les droits de Nîmes'}`}
   410→                      className={cn(
   411→                        'shrink-0 rounded-full px-2 py-0.5 text-[10px] font-bold uppercase tracking-wider transition-colors',
   412→                        m.city === 'NIMES'
   413→                          ? 'bg-primary/10 text-primary hover:bg-primary/20'
   414→                          : 'bg-accent/15 text-gold-vif hover:bg-accent/25'
   415→                      )}
   416→                    >
   417→                      {VILLE_LABEL[m.city] ?? m.city}
   418→                    </button>
   419→                    {lireSurcharges(m.perms) &&
   420→                      Object.keys(lireSurcharges(m.perms)).length > 0 && (
   421→                        <span
   422→                          title="Ce frère porte des droits particuliers"
   423→                          className="shrink-0 rounded-full bg-primary/12 px-2 py-0.5 text-[9px] font-bold uppercase tracking-wider text-primary"
   424→                        >
   425→                          <SlidersHorizontal className="mr-0.5 inline h-2.5 w-2.5" />
   426→                          droits
   427→                        </span>
   428→                      )}
   429→                    {m.status === 'BLOCKED' && (
   430→                      <span className="shrink-0 rounded-full bg-destructive/12 px-2 py-0.5 text-[9px] font-bold uppercase tracking-wider text-destructive">
   431→                        {STATUT_LABEL[m.status]}
   432→                      </span>
   433→                    )}
   434→                  </span>
   435→                  <div className="flex shrink-0 flex-wrap gap-1.5">
   436→                    {(
   437→                      [
   438→                        { role: 'MEMBER', label: 'Membre' },
   439→                        { role: 'TREASURER', label: 'Trésorier' },
   440→                        { role: 'REFERENT', label: 'Référent' },
   441→                      ] as { role: string; label: string }[]
   442→                    ).map((r) => (
   443→                      <button
   444→                        key={r.role}
   445→                        onClick={() => majMembre.mutate({ id: m.id, role: r.role })}
   446→                        className={cn(
   447→                          'rounded-full px-2.5 py-1 text-[10px] font-bold uppercase tracking-wider transition-colors',
   448→                          m.role === r.role
   449→                            ? r.role === 'REFERENT'
   450→                              ? 'bg-primary text-white'
   451→                              : r.role === 'TREASURER'
   452→                                ? 'bg-accent text-white dark:text-accent-foreground'
   453→                                : 'bg-primary/10 text-primary'
   454→                            : 'bg-sand text-muted-foreground hover:bg-primary/10 hover:text-primary'
   455→                        )}
   456→                        title={
   457→                          r.role === 'TREASURER' && m.role !== 'TREASURER'
   458→                            ? `Nommer trésorier (remplace l'actuel)`
   459→                            : r.role === 'REFERENT' && m.role !== 'REFERENT' && referent
   460→                              ? `Remplacer ${referent.name} comme référent`
   461→                              : r.label
   462→                        }
   463→                      >
   464→                        {r.role === 'TREASURER' && <Wallet className="mr-1 inline h-3 w-3" />}
   465→                        {r.role === 'REFERENT' && <BadgeCheck className="mr-1 inline h-3 w-3" />}
   466→                        {r.label}
   467→                      </button>
   468→                    ))}
   469→                  </div>
   470→                </div>
   471→
   472→                {/* Ligne d'accès : droits particuliers, code, révocation */}
   473→                <div className="flex flex-wrap items-center gap-1.5 pl-6">
   474→                  <button
   475→                    onClick={() => setMembreDroits(m)}
   476→                    className="inline-flex items-center gap-1 rounded-full bg-sand px-2.5 py-1 text-[10px] font-bold uppercase tracking-wider text-muted-foreground transition-colors hover:bg-primary/10 hover:text-primary"
   477→                    title="Accorder ou retirer des droits précis à ce frère"
   478→                  >
   479→                    <SlidersHorizontal className="h-3 w-3" /> Droits
   480→                  </button>
   481→                  <button
   482→                    onClick={() =>
   483→                      majMembre.mutate(
   484→                        { id: m.id, resetPin: true },
   485→                        {
   486→                          onSuccess: () =>

   609→/* ═══════════════ Droits particuliers d'un frère ═══════════════ */
   610→function DroitsDialog({
   611→  membre,
   612→  onOpenChange,
   613→}: {
   614→  membre: Member | null
   615→  onOpenChange: (v: boolean) => void
   616→}) {
   617→  const qc = useQueryClient()
   618→  const { toast } = useToast()
   619→  const open = membre !== null
   620→
   621→  const majMembre = useMutation({
   622→    mutationFn: ({ id, perms }: { id: string; perms: Record<string, boolean> | null }) =>
   623→      api(`/api/members/${id}`, { method: 'PATCH', body: JSON.stringify({ perms }) }),
   624→    onSuccess: () => {
   625→      qc.invalidateQueries({ queryKey: ['members'] })
   626→    },
   627→  })
   628→
   629→  /* État local des surcharges — synchronisé à l'ouverture */
   630→  const [surcharges, setSurcharges] = useState<Surcharges>({})
   631→  const [initialise, setInitialise] = useState(false)
   632→  if (open && !initialise && membre) {
   633→    setSurcharges(lireSurcharges(membre.perms))
   634→    setInitialise(true)
   635→  }
   636→  if (!open && initialise) setInitialise(false)
   637→
   638→  if (!membre) return null
   639→
   640→  const appliquer = (nouvelles: Surcharges) => {
   641→    setSurcharges(nouvelles)
   642→    majMembre.mutate(
   643→      { id: membre.id, perms: Object.keys(nouvelles).length ? nouvelles : null },
   644→      {
   645→        onError: () =>
   646→          toast({
   647→            title: 'Impossible d’enregistrer',
   648→            description: 'Réessaie dans un instant.',
   649→            variant: 'destructive',
   650→          }),
   651→      }
   652→    )
   653→  }
   654→
   655→  const basculer = (famille: FamilleDroits, valeur: boolean) => {
   656→    const defaut = peut(membre.city, 'MEMBER', DROIT_DE_LA_FAMILLE[famille])
   657→    const nouvelles = { ...surcharges }
   658→    if (valeur === defaut) {
   659→      /* Retour au droit du groupe — la surcharge disparaît */
   660→      delete nouvelles[famille]
   661→    } else {
   662→      nouvelles[famille] = valeur
   663→    }
   664→    appliquer(nouvelles)
   665→  }
   666→
   667→  const nombreSurcharges = Object.keys(surcharges).length
   668→
   669→  return (
   670→    <Dialog open={open} onOpenChange={onOpenChange}>
   671→      <DialogContent className="max-w-md rounded-3xl border-primary/15 bg-ivoire p-6">
   672→        <DialogHeader>
   673→          <DialogTitle className="flex items-center gap-2 font-display text-2xl text-encre">
   674→            <SlidersHorizontal className="h-5 w-5 text-primary" /> Droits de Sidi {membre.name}
   675→          </DialogTitle>
   676→          <DialogDescription>
   677→            {VILLE_LABEL[membre.city] ?? membre.city} · {ROLE_LABEL[membre.role] ?? 'Membre'} —
   678→            accorde ou retire des droits précis, au-dessus de ceux de son groupe.
   679→          </DialogDescription>
   680→        </DialogHeader>
   681→
   682→        {membre.role === 'REFERENT' ? (
   683→          <p className="rounded-xl bg-sand/60 px-3 py-3 text-sm leading-relaxed text-muted-foreground">
   684→            Le référent porte toutes les charges de la zaouia — il peut déjà tout faire.
   685→          </p>
   686→        ) : (
   687→          <div className="space-y-2">
   688→            {FAMILLES_DROITS.map((f) => {
   689→              const defaut = peut(membre.city, 'MEMBER', DROIT_DE_LA_FAMILLE[f.id])
   690→              const particuliere = f.id in surcharges
   691→              const effective = particuliere
   692→                ? surcharges[f.id] === true
   693→                : peut(membre.city, membre.role, DROIT_DE_LA_FAMILLE[f.id], surcharges)
   694→              return (
   695→                <div
   696→                  key={f.id}
   697→                  className={cn(
   698→                    'flex items-center gap-3 rounded-2xl border px-4 py-3 transition-colors',
   699→                    particuliere
   700→                      ? effective
   701→                        ? 'border-primary/30 bg-primary/6'
   702→                        : 'border-accent/35 bg-or-pale/70'
   703→                      : 'border-primary/10 bg-card'
   704→                  )}
   705→                >
   706→                  <div className="min-w-0 flex-1">
   707→                    <p className="text-sm font-semibold text-secondary-foreground">{f.label}</p>
   708→                    <p className="text-[11px] leading-snug text-brouillard">{f.detail}</p>
   709→                    <p
   710→                      className={cn(
   711→                        'mt-1 text-[10px] font-bold uppercase tracking-wider',
   712→                        particuliere
   713→                          ? effective
   714→                            ? 'text-primary'
   715→                            : 'text-gold-vif'
   716→                          : 'text-brouillard-clair'
   717→                      )}
   718→                    >
   719→                      {particuliere
   720→                        ? effective
   721→                          ? 'droit particulier accordé'
   722→                          : 'droit retiré (particulier)'
   723→                        : `droit du groupe — ${defaut ? 'oui' : 'non'}`}
   724→                    </p>
   725→                  </div>
   726→                  <Switch
   727→                    checked={effective}
   728→                    onCheckedChange={(v) => basculer(f.id, v)}
   729→                    aria-label={`${f.label} — ${effective ? 'accordé' : 'refusé'}`}
   730→                  />
   731→                </div>
   732→              )
   733→            })}
   734→
   735→            {nombreSurcharges > 0 && (
   736→              <Button
   737→                variant="outline"
   738→                size="sm"
   739→                onClick={() => appliquer({})}
   740→                className="w-full rounded-full border-dashed border-primary/40 text-primary"
   741→              >
   742→                Revenir aux droits du groupe
   743→              </Button>
   744→            )}
   745→            <p className="text-center text-[10px] leading-relaxed text-brouillard-clair">
   746→              Un interrupteur sur sa position de groupe fait disparaître le réglage particulier.
   747→            </p>
   748→          </div>
   749→        )}
   750→
   751→        <DialogFooter>
   752→          <Button
   753→            onClick={() => onOpenChange(false)}
   754→            className="w-full rounded-full bg-primary text-white hover:bg-emerald-deep"
   755→          >
   756→            Terminé
   757→          </Button>
   758→        </DialogFooter>
   759→      </DialogContent>
   760→    </Dialog>
   761→  )
   762→}

   765→export function ProfileDialog({
   766→  open,
   767→  onOpenChange,
   768→}: {
   769→  open: boolean
   770→  onOpenChange: (v: boolean) => void
   771→}) {
   772→  const name = useUser((s) => s.name)
   773→  const ville = useUser((s) => s.ville)
   774→  const setName = useUser((s) => s.setName)
   775→  const setMember = useUser((s) => s.setMember)
   776→  const logout = useUser((s) => s.logout)
   777→  const { data: membres } = useMembers()
   778→  const { toast } = useToast()
   779→  const [choixManuel, setChoixManuel] = useState<string | null>(null)
   780→  const [erreur, setErreur] = useState<string | null>(null)
   781→  const choix = choixManuel ?? name ?? ''
   782→
   783→  /* Les frères de MA zawiya — on y cherche, on y entre */
   784→  const freresVille = (membres ?? []).filter((m) => m.city === ville && m.status === 'ACTIVE')
   785→  const monMembre: Member | undefined = freresVille.find((m) => m.name === name)
   786→  const cestMoi = monMembre !== undefined && choix.trim() === monMembre.name
   787→
   788→  const valider = () => {
   789→    const v = choix.trim()
   790→    if (!v) return
   791→
   792→    /* Rester soi-même — rien à faire */
   793→    if (cestMoi) {
   794→      onOpenChange(false)
   795→      return
   796→    }
   797→
   798→    /* La porte est ouverte : on entre sans code — la sécurité
   799→       reviendra avec les champs de la porte d'entrée. */
   800→    const frere = freresVille.find((m) => m.name === v)
   801→    if (!frere) {
   802→      setErreur('Ce prénom n’est pas inscrit à la zawiya.')
   803→      return
   804→    }
   805→    setName(frere.name)
   806→    setMember(frere.id, frere.role, frere.city, frere.perms)
   807→    setChoixManuel(null)
   808→    setErreur(null)
   809→    toast({ title: `Marhaban bik, Sidi ${frere.name}` })
   810→    onOpenChange(false)
   811→  }
   812→
   813→  return (
   814→    <Dialog open={open} onOpenChange={onOpenChange}>
   815→      <DialogContent className="max-w-sm rounded-3xl border-primary/15 bg-ivoire p-6">
   816→        <DialogHeader>
   817→          <DialogTitle className="font-display text-2xl text-encre">Je suis…</DialogTitle>
   818→          <DialogDescription>
   819→            {monMembre
   820→              ? `Sidi ${monMembre.name} — ${ROLE_LABEL[monMembre.role] ?? 'membre'} de la zawiya de ${VILLE_LABEL[monMembre.city] ?? 'Nîmes'}`
   821→              : `Indique ton prénom pour la zawiya de ${VILLE_LABEL[ville] ?? 'Nîmes'}.`}
   822→          </DialogDescription>
   823→        </DialogHeader>
   824→
   825→        <div className="space-y-3">
   826→          <div className="space-y-1.5">
   827→            <Label className="text-[11px] font-bold uppercase tracking-[0.25em] text-muted-foreground">
   828→              Prénom
   829→            </Label>
   830→            <div className="relative">
   831→              <input
   832→                value={choix}
   833→                onChange={(e) => setChoixManuel(e.target.value)}
   834→                list="prenoms-zaouia"
   835→                placeholder="Ton prénom…"
   836→                className="w-full rounded-full border border-primary/25 bg-card py-3 pl-5 pr-5 text-center text-lg text-encre outline-none transition-all focus:border-primary focus:ring-4 focus:ring-primary/10"
   837→              />
   838→              <datalist id="prenoms-zaouia">
   839→                {freresVille.map((m) => (
   840→                  <option key={m.id} value={m.name} />
   841→                ))}
   842→              </datalist>
   843→            </div>
   844→          </div>
   845→
   846→          {/* Suggestions rapides — les frères de sa zawiya */}
   847→          <div className="flex flex-wrap justify-center gap-1.5">
   848→            {freresVille.slice(0, 8).map((m) => (
   849→              <button
   850→                key={m.id}
   851→                onClick={() => setChoixManuel(m.name)}
   852→                className={cn(
   853→                  'rounded-full px-3 py-1.5 text-xs font-semibold transition-colors',
   854→                  choix === m.name
   855→                    ? 'bg-primary text-white'
   856→                    : 'bg-card text-primary ring-1 ring-primary/15 hover:bg-primary/5'
   857→                )}
   858→              >
   859→                Sidi {m.name}

   861→            ))}
