    87→/* ─── Une ligne du texte, répétitions « x 100 » / « × ١٠٠ » en badge ─── */
    88→function LigneInvocation({ ligne, accent }: { ligne: string; accent: string }) {
    89→  const m = ligne.match(/^(.*?)(\s*[×x]\s*[٠-٩\dHh '’]+)$/u)
    90→  if (m && m[2].trim()) {
    91→    return (
    92→      <span className="block">
    93→        {m[1]}
    94→        <span
    95→          className="mx-1 inline-block rounded-md px-2 py-0.5 align-middle text-[0.72em] font-bold tracking-wide"
    96→          style={{ color: accent, backgroundColor: `${accent}18` }}
    97→        >
    98→          {m[2].trim()}
    99→        </span>
   100→      </span>
   101→    )
   102→  }
   103→  return <span className="block">{ligne || '\u00A0'}</span>
   104→}

   186→function VersPoeme({ ligne, accent }: { ligne: string; accent: string }) {
   187→  const coupe = ligne.split('۞')
   188→  if (coupe.length !== 2) {
   189→    return <LigneInvocation ligne={ligne} accent={accent} />
   190→  }
   191→  return (
   192→    <span className="poesie-vers">
   193→      <span className="poesie-shatr">{coupe[0].trim()}</span>
   194→      <span aria-hidden className="poesie-fasl" style={{ color: accent }}>
   195→        ۞
   196→      </span>
   197→      <span className="poesie-shatr">{coupe[1].trim()}</span>
   198→    </span>
   199→  )
   200→}

   202→/* ─── Le paragraphage du cahier phonétique ───
   203→   Le cahier imprimé est une colonne étroite : l'extraction garde
   204→   des lignes courtes coupées au milieu des phrases. On rejoint
   205→   les lignes d'un même bloc (séparé par une ligne vide) et on
   206→   coupe après les renvois « → p. » et les répétitions « x 100 » —
   207→   le texte coule alors sur toute la largeur, justifié comme un
   208→   livre, au lieu de rester collé à gauche. */
   209→function paragrapher(texte: string): string[] {
   210→  const paras: string[] = []
   211→  let courant: string[] = []
   212→  const flush = () => {
   213→    if (courant.length) {
   214→      paras.push(courant.join(' ').replace(/\s+/g, ' ').trim())
   215→      courant = []
   216→    }
   217→  }
   218→  for (const brute of texte.split('\n')) {
   219→    const t = brute.trim()
   220→    if (!t) {
   221→      flush()
   222→      continue
   223→    }
   224→    courant.push(t)
   225→    /* Fin logique de paragraphe : renvoi de page, ou répétition badgée */
   226→    if (/→\s*p\.|→\s*ص|[×x]\s*[٠-٩\d][٠-٩\dHh '’]*$/.test(t)) flush()
   227→  }
   228→  flush()
   229→  return paras
   230→}

   251→function LecteurAdhkar({ doc, langue, chapitreInitial, onFermer }: LecteurProps) {
   252→  const l = useLecture()
   253→  const imprimer = usePrint((s) => s.imprimer)
   254→
   255→  /* Le diwân de samâ‘ (398 chants) n'entre pas dans le fil principal :
   256→     il se charge paresseusement à l'ouverture du document. */
   257→  const estSama = doc.id === 'sama' && langue === 'ph'
   258→  const [samaTexte, setSamaTexte] = useState<ChapitreLecture[] | null>(null)
   259→  useEffect(() => {
   260→    if (doc.id !== 'sama' || langue !== 'ph') return
   261→    let annule = false
   262→    import('@/data/sama-chants').then((m) => {
   263→      if (!annule) setSamaTexte(m.CHAPITRES_SAMA)
   264→    })
   265→    return () => {
   266→      annule = true
   267→    }
   268→  }, [doc.id, langue])
   269→
   270→  /* Les entrées du lecteur : pratiques structurées (cahier de
   271→     wadhifa) ou chapitres plats (autres documents). Chaque entrée
   272→     porte sa langue effective — le sommaire ne change jamais quand
   273→     on bascule phonétique ⇄ عربي, la pratique lue reste en place. */
   274→  const entrees = useMemo<EntreeLecture[]>(() => {
   275→    if (doc.pratiques) {
   276→      return doc.pratiques
   277→        .map((p): EntreeLecture | null => {
   278→          const prefere = p[langue]
   279→          const autre = langue === 'ph' ? p.ar : p.ph
   280→          const c = prefere ?? autre
   281→          if (!c) return null
   282→          return {
   283→            ...c,
   284→            langue: prefere ? langue : langue === 'ph' ? 'ar' : 'ph',
   285→            repli: !prefere,
   286→            groupe: p.groupe,
   287→          }
   288→        })
   289→        .filter((e): e is EntreeLecture => e !== null)
   290→    }
   291→    if (doc.id === 'sama' && langue === 'ph') {
   292→      return (samaTexte ?? []).map((c) => ({ ...c, langue: 'ph' as const }))
   293→    }
   294→    return chapitresDuDoc(doc, langue).map((c) => ({ ...c, langue }))
   295→  }, [doc, langue, samaTexte])
   296→  const chapitres = entrees
   297→  const pdf = pdfDuDoc(doc, langue)
   298→  const conteneurRef = useRef<HTMLDivElement | null>(null)
   299→  const cartesRef = useRef<(HTMLElement | null)[]>([])
   300→
   301→  const [indexActif, setIndexActif] = useState(() => {
   302→    const i = chapitreInitial ? entrees.findIndex((c) => c.id === chapitreInitial) : -1
   303→    return i >= 0 ? i : 0
   304→  })
   305→  const [reglagesOuverts, setReglagesOuverts] = useState(false)
   306→  const [sommaireOuvert, setSommaireOuvert] = useState(false)
   307→  /* La carte sommaire du cahier — dépliée à l'ouverture, refermée
   308→     si la recherche ⌘K demande directement une pratique. */
   309→  const [sommaireCarte, setSommaireCarte] = useState(!chapitreInitial)
   310→  const [progression, setProgression] = useState(0)
   311→
   312→  const arabe = langue === 'ar'
   313→  /* Vue du lecteur : le TEXTE RÉÉCRIT d'abord — encre sur porcelaine,
   314→     polices et couleurs réglables, évolutif. La page numérisée reste
   315→     disponible pour lire le livre imprimé à l'identique. La recherche ⌘K
   316→     ouvre directement le texte au chapitre demandé. */
   317→  const aLesDeux = Boolean(pdf) && chapitres.length > 0
   318→  const [vue, setVue] = useState<'doc' | 'texte'>(
   319→    chapitreInitial || !pdf || chapitres.length > 0 || estSama ? 'texte' : 'doc'
   320→  )
   321→  const vueDoc = vue === 'doc' && Boolean(pdf)
   322→  const theme = THEMES.find((t) => t.id === l.theme) ?? THEMES[0]
   323→  const accent = ACCENTS.find((a) => a.id === l.accent) ?? ACCENTS[0]
   324→  const police = arabe
   325→    ? POLICES_AR.find((p) => p.id === l.policeArabe) ?? POLICES_AR[0]
   326→    : POLICES_PH.find((p) => p.id === l.police) ?? POLICES_PH[0]
   327→  const taille = arabe ? l.tailleArabe : l.taille
   328→
   329→  /* ─── L'écran ne s'éteint pas pendant qu'on lit ───
   330→     Un frère qui récite ne touche pas son téléphone pendant de
   331→     longues minutes. Le verrou d'écran est demandé à l'ouverture,
   332→     rendu si la page passe à l'arrière-plan, et relâché après un
   333→     quart d'heure sans le moindre geste — un téléphone oublié
   334→     écran allumé chauffe. Tout échec est muet. */
   335→  useEffect(() => {
   336→    if (!('wakeLock' in navigator)) return
   337→    let verrou: WakeLockSentinel | null = null
   338→    let minuteur: number | undefined
   339→    const REPOS = 15 * 60 * 1000
   340→
   341→    const demander = async () => {
   342→      try {
   343→        verrou = await navigator.wakeLock.request('screen')
   344→      } catch {
   345→        /* appareil qui ne sait pas faire : on lit comme avant */
   346→      }
   347→    }
   348→    const relacher = () => {
   349→      verrou?.release().catch(() => undefined)
   350→      verrou = null
   351→    }
   352→    const toucher = () => {
   353→      window.clearTimeout(minuteur)
   354→      minuteur = window.setTimeout(relacher, REPOS)
   355→    }
   356→    const surVisibilite = () => {
   357→      if (document.visibilityState === 'visible') {
   358→        demander()
   359→        toucher()
   360→      } else {
   361→        relacher()
   362→      }
   363→    }
   364→    demander()
   365→    toucher()
   366→    document.addEventListener('visibilitychange', surVisibilite)
   367→    window.addEventListener('pointerdown', toucher, { passive: true })
   368→    return () => {
   369→      relacher()
   370→      window.clearTimeout(minuteur)
   371→      document.removeEventListener('visibilitychange', surVisibilite)
   372→      window.removeEventListener('pointerdown', toucher)
   373→    }
   374→  }, [])
   375→
   376→  /* ─── Le corps de la page ne défile pas derrière le lecteur ─── */
   377→  useEffect(() => {
   378→    const deplacement = window.scrollY
   379→    document.body.style.overflow = 'hidden'
   380→    return () => {
   381→      document.body.style.overflow = ''
   382→      window.scrollTo(0, deplacement)
   383→    }
   384→  }, [])
   385→
   386→  /* ─── Échap ferme le lecteur ─── */
   387→  useEffect(() => {
   388→    const surTouche = (e: KeyboardEvent) => {
   389→      if (e.key === 'Escape') onFermer()
   390→    }
   391→    window.addEventListener('keydown', surTouche)
   392→    return () => window.removeEventListener('keydown', surTouche)
   393→  }, [onFermer])
   394→
   395→  /* ─── Progression de la lecture ─── */
   396→  useEffect(() => {
   397→    const el = conteneurRef.current
   398→    if (!el) return
   399→    const surScroll = () => {
   400→      const reste = el.scrollHeight - el.clientHeight
   401→      setProgression(reste > 0 ? Math.min(1, el.scrollTop / reste) : 0)
   402→    }
   403→    el.addEventListener('scroll', surScroll, { passive: true })
   404→    surScroll()
   405→    return () => el.removeEventListener('scroll', surScroll)
   406→  }, [vueDoc])
   407→
   408→  /* ─── Positionner au chapitre demandé à l'ouverture ───
   409→     (le diwân arrive un instant après : on replace alors)
   410→     « langue » n'est PAS une dépendance : basculer phonétique ⇄ عربي
   411→     ne doit jamais replacer la lecture — les pratiques unifiées
   412→     gardent leur place, seul le texte affiché change. */
   413→  useEffect(() => {
   414→    const el = conteneurRef.current
   415→    if (!el) return
   416→    if (chapitreInitial) {
   417→      const i = chapitres.findIndex((c) => c.id === chapitreInitial)
   418→      const cible = i >= 0 ? cartesRef.current[i] : null
   419→      if (cible) {
   420→        cible.scrollIntoView({ block: 'start' })
   421→        return
   422→      }
   423→    }
   424→    el.scrollTo({ top: 0 })
   425→  }, [chapitreInitial, doc.id, chapitres.length])
   426→
   427→  /* Ancre robuste : scrollIntoView + marge de défilement — le
   428→     navigateur mesure lui-même la position réelle, quel que soit
   429→     l'état de l'en-tête ou des panneaux déroulés au-dessus. */
   430→  const allerAuChapitre = useCallback((i: number) => {
   431→    const cible = cartesRef.current[i]
   432→    if (cible) cible.scrollIntoView({ behavior: 'smooth', block: 'start' })
   433→  }, [])
   434→
   435→  /* ─── Le chapitre visible suit le défilement ─── */
   436→  useEffect(() => {
   437→    if (chapitres.length < 2) return
   438→    const el = conteneurRef.current
   439→    if (!el) return
   440→    const observateur = new IntersectionObserver(
   441→      (entrees) => {
   442→        for (const e of entrees) {
   443→          if (e.isIntersecting) {
   444→            const i = Number((e.target as HTMLElement).dataset.i)
   445→            if (!Number.isNaN(i)) setIndexActif(i)
   446→          }
   447→        }
   448→      },
   449→      { root: el, rootMargin: '-15% 0px -60% 0px', threshold: 0 }
   450→    )
   451→    cartesRef.current.forEach((c) => c && observateur.observe(c))
   452→    return () => observateur.disconnect()
   453→  }, [chapitres.length, langue, doc.id])
   454→
   455→  const titre = arabe ? doc.titreAr : doc.titreFr
   456→
   457→  return (
   458→    <motion.div
   459→      initial={{ opacity: 0 }}
   460→      animate={{ opacity: 1 }}
   461→      exit={{ opacity: 0 }}
   462→      transition={{ duration: 0.22 }}
   463→      /* mb-0 : parade contre les space-y-* d'un ancêtre (Tailwind v4
   464→          pose margin-block-end sur :not(:last-child)) — un lecteur plein
   465→          écran ne doit jamais recevoir de marge, où qu'il vive */
   466→      className="fixed inset-0 z-[90] mb-0 flex flex-col bg-background grain-papier"
   467→      role="dialog"
   468→      aria-modal="true"
   469→      aria-label={`Lecteur — ${titre}`}
   470→    >
   471→      {/* ─── La barre du haut : titre, sommaire, réglages, fermeture ─── */}
   472→      <header className="relative z-10 flex-none border-b border-border bg-card/95 backdrop-blur supports-[backdrop-filter]:bg-card/90">
   473→        <div
   474→          className="flex items-center gap-1 px-2.5"
   475→          style={{ paddingTop: 'calc(10px + env(safe-area-inset-top))', paddingBottom: 10 }}
   476→        >
   477→          {/* Réglages de lecture — le texte seulement */}
